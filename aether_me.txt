    function strategicDistanceScale() {
        let width = 0;
        let height = 0;
        let diagonal = 0;

        width = mapWidth();
        height = mapHeight();
        diagonal = width + height;

        if (diagonal < 300) {
            diagonal = 300;

    }

    function updateIncomeModel() {
        let now = 0;
        let gathered = 0;
        let elapsed = 0;
        let gained = 0;

        now = gameTime();

        if (now < economyTick) {
            return;
        }

        gathered = resourcesGathered();

        if (lastGatheredTime <= 0) {
            lastGatheredTime = now;
            lastGatheredSnapshot = gathered;
            incomeRate = 0;
            economyTick = now + 5;
            return;
        }

        elapsed = now - lastGatheredTime;
        gained = gathered - lastGatheredSnapshot;

        if (elapsed > 0) {
            incomeRate = gained / elapsed;
        }

        lastGatheredTime = now;
        lastGatheredSnapshot = gathered;
        economyTick = now + 7;
    }

    function updateLossModel() {
        let now = 0;
        let delta = 0;
        now = gameTime();
        delta = 0;
        if (now < attritionTick) { return; }
        // Only actual main-army deaths affect attack/recovery decisions.
        for (id in observedArmy) {
            if (!unitExists(id)) { delta = delta + 1; }
        }
        observedArmy = mainFighters();
        if (lossWindowTime <= 0 || now - lossWindowTime > 35) {
            recentLosses = 0;
        }
        if (delta > 0) {
            recentLosses = recentLosses + delta;
            lossWindowTime = now;
        }
        attritionTick = now + 3;
    }

    function calculateArmyHealth() {
        let totalRatio = 0;
        let number = 0;
        let damaged = 0;
        let healthy = 0;

        totalRatio = 0;
        number = 0;
        damaged = 0;
        healthy = 0;

        for (id in mainFighters()) {
            if (unitExists(id)) {
                totalRatio = totalRatio + unitHealthRatio(id);
                number = number + 1;

                if (unitHealthRatio(id) < 0.48) {
                    damaged = damaged + 1;
                } else {
                    healthy = healthy + 1;
                }
            }
        }

        damagedArmyCount = damaged;
        healthyArmyCount = healthy;

        if (number <= 0) {
            armyHealthScore = 1;
            return 1;
        }

        armyHealthScore = totalRatio / number;
        return armyHealthScore;
    }

    function powerOfEnemyList(list) {
        let power = 0;
        let hpWeight = 0;
        let rangeWeight = 0;

        power = 0;

        for (enemy in list) {
            if (unitExists(enemy)) {
                hpWeight = unitHealthRatio(enemy);
                rangeWeight = unitAttackRange(enemy);

                if (rangeWeight > 70) {
                    power = power + unitAttack(enemy) + 8;
                } else {
                    power = power + unitAttack(enemy) + 3;
                }

                if (hpWeight > 0.75) {
                    power = power + 4;
                }
            }
        }

        return power;
    }

    function powerOfFriendlyList(list) {
        let power = 0;

        power = totalAttack(list);
        power = power + count(list) * 3;

        return power;
    }

    function updateLocalBattleModel() {
        let centerX = 0;
        let centerY = 0;
        let anchor = "";
        let friends = [];
        let contacts = [];
        let seen = false;
        let enemies = 0;
        let weight = 0;
        centerX = baseX();
        centerY = baseY();
        friends = [];
        contacts = [];
        enemies = 0;
        anchor = closestUnitTo(mainFighters(), enemyBaseX(), enemyBaseY());
        if (phase == 2 && marchingAnchor != "" && unitExists(marchingAnchor)) {
            anchor = marchingAnchor;
        }
        if (anchor != "" && unitExists(anchor)) {
            centerX = unitX(anchor);
            centerY = unitY(anchor);
        }
        for (id in mainFighters()) {
            if (unitExists(id) && inRange(unitX(id), unitY(id), centerX, centerY, 155)) {
                friends[count(friends)] = id;
            }
        }
        // Keep unseen contacts briefly; vanishing into fog is not a confirmed kill.
        for (contact in recentEnemyContacts) {
            seen = false;
            for (enemy in visibleEnemyUnits()) {
                if (enemy == contact[0]) { seen = true; }
            }
            if (!seen && gameTime() - contact[4] < 12) {
                contacts[count(contacts)] = contact;
            }
        }
        for (enemy in visibleEnemyUnits()) {
            if (unitExists(enemy) && unitAttack(enemy) > 0) {
                weight = unitAttack(enemy) + 3;
                if (unitAttackRange(enemy) > 70) { weight = weight + 5; }
                if (unitHealthRatio(enemy) > 0.75) { weight = weight + 4; }
                contacts[count(contacts)] = [enemy, unitX(enemy), unitY(enemy), weight, gameTime()];
            }
        }
        recentEnemyContacts = contacts;
        for (contact in recentEnemyContacts) {
            if (inRange(contact[1], contact[2], centerX, centerY, 200)) {
                enemies = enemies + contact[3];
            }
        }
        friendlyLocalPower = powerOfFriendlyList(friends);
        enemyLocalPower = enemies;
        if (enemies <= 0) { localPowerRatio = 3; }
        else { localPowerRatio = friendlyLocalPower / enemies; }

        // A momentary bad ratio means hold; sustained disadvantage means retreat.
        if (phase == 2 && enemies > 0 && localPowerRatio < 0.68) {
            if (losingFightSince < 0) { losingFightSince = gameTime(); }
        } else { losingFightSince = -1; }
        armyFightRetreat = losingFightSince >= 0 && gameTime() - losingFightSince >= 3;
    }

    function updateGlobalBattleModel() {
        friendlyGlobalPower = calculateMainPower();
        enemyGlobalPower = powerOfEnemyList(visibleEnemyUnits());

        if (enemyGlobalPower <= 0) {
            globalPowerRatio = 3;
        } else {
            globalPowerRatio = friendlyGlobalPower / enemyGlobalPower;
        }

        pursuitMode = (
            phase == 2 &&
            globalPowerRatio >= 1.8 &&
            armyHealthScore >= 0.6
        );
    }

    function computeResourceReserve() {
        let reserve = 0;
        let uncoveredNeed = 0;
        let defensiveNeed = 0;
        let productionNeed = 0;

        reserve = 120;
        uncoveredNeed = desiredOutposts() - buildingCount("Whisper Node");
        defensiveNeed = 0;
        productionNeed = 0;

        if (!hasBuilding("Spawning Vault")) {
            productionNeed = productionNeed + 170;
        }

        if (
            hasBuilding("Spawning Vault") &&
            !hasBuilding("Titan Crucible") &&
            gameTime() > 45
        ) {
            productionNeed = productionNeed + 220;
        }

        if (uncoveredNeed > 0) {
            reserve = reserve + 180;
        }

        if (commandCenterUnderAttack()) {
            defensiveNeed = defensiveNeed + 200;
        }

        if (emergencyActive) {
            defensiveNeed = defensiveNeed + 140;
        }

        if (unitCount("Shaper") < desiredShapers()) {
            reserve = reserve + 65;
        }

        if (incomeRate < 3 && gameTime() > 50) {
            reserve = reserve + 100;
        }

        if (recentLosses >= 4) {
            reserve = reserve + 100;
        }

        reserve = reserve + defensiveNeed + productionNeed;
        reserveResources = reserve;

        return reserve;
    }

    function updateStrategicModes() {
        let now = 0;
        let army = 0;
        let workers = 0;

        now = gameTime();

        if (now < strategicTick) {
            return;
        }

        mapDiagonal = strategicDistanceScale();
        enemyBaseDistance = distanceBetween(
            baseX(),
            baseY(),
            enemyBaseX(),
            enemyBaseY()
        );

        updateIncomeModel();
        updateLossModel();
        calculateArmyHealth();
        updateLocalBattleModel();
        computeResourceReserve();

        army = mainArmyCount();
        workers = unitCount("Shaper");
        enemyPressure = baseThreat + workerThreat + assetThreat;

        economyMode = 1;

        if (workers < 3) {
            economyMode = 2;
        }

        if (incomeRate < 2 && now > 65) {
            economyMode = 2;
        }

        if (enemyPressure > 0 || recentLosses >= 5) {
            economyMode = 0;
        }

        pressureMode = 0;

        if (army >= requiredArmy && armyHealthScore > 0.70) {
            pressureMode = 1;
        }

        if (localPowerRatio > 1.65 && army >= 8) {
            pressureMode = 2;
        }

        if (
            enemyPressure > 0 ||
            armyHealthScore < 0.52 ||
            recentLosses >= 6
        ) {
            pressureMode = -1;
        }

        safeToSpend = resources() > reserveResources;
        safeToExpand = (
            safeToSpend &&
            enemyPressure <= 0 &&
            pressureMode >= 0 &&
            unitCount("Shaper") >= 4
        );
        safeToResearch = (
            resources() > reserveResources + 130 &&
            !emergencyActive &&
            recentLosses < 4
        );
        safeToTech = (
            resources() > reserveResources + 200 &&
            mainArmyCount() >= 6 &&
            !emergencyActive
        );

        desiredFrontline = max(4, floor(army * 0.34));
        desiredRanged = max(4, floor(army * 0.40));
        desiredSupport = max(1, floor(army * 0.12));
        desiredAir = max(0, floor(army * 0.18));
        desiredNaval = 0;

        if (waterAvailable) {
            desiredNaval = max(2, floor(army * 0.20));
        }

        aggressionBudget = resources() - reserveResources;
        strategicTick = now + 2;
    }

    function nodeThreatValue(node) {
        let value = 0;
        let nx = 0;
        let ny = 0;
        let distanceHome = 0;
        let distanceEnemy = 0;
        let enemies = 0;

        if (node == "" || !nodeExists(node)) {
            return 9999;
        }

        nx = nodeX(node);
        ny = nodeY(node);
        distanceHome = distanceBetween(baseX(), baseY(), nx, ny);
        distanceEnemy = distanceBetween(enemyBaseX(), enemyBaseY(), nx, ny);
        enemies = count(enemyUnitsNear(nx, ny, 180));

        value = enemies * 100;

        if (enemyBaseDistance >= 120 && distanceEnemy < distanceHome) {
            value = value + 80;
        }

        if (enemyBaseDistance >= 120 && distanceEnemy < 220) {
            value = value + 160;
        }

        if (distanceHome > mapDiagonal * 0.42) {
            value = value + 70;
        }

        if (nodeRemaining(node) < 250) {
            value = value + 120;
        }

        return value;
    }

    function nodeOnFriendlySide(node) {
        let homeDistance = 0;
        let enemyDistance = 0;

        if (node == "" || !nodeExists(node)) {
            return false;
        }

        homeDistance = distanceBetween(
            baseX(),
            baseY(),
            nodeX(node),
            nodeY(node)
        );
        enemyDistance = distanceBetween(
            enemyBaseX(),
            enemyBaseY(),
            nodeX(node),
            nodeY(node)
        );

        if (enemyBaseDistance < 120) {
            return nodeHasEconomicConnection(node);
        }

        return homeDistance + expansionSafetyMargin < enemyDistance;
    }

    function nodeHasEnemyPresence(node) {
        if (node == "" || !nodeExists(node)) {
            return true;
        }

        if (
            count(
                enemyUnitsNear(
                    nodeX(node),
                    nodeY(node),
                    210
                )
            ) > 0
        ) {
            return true;
        }

        if (
            count(
                enemyBuildingsNear(
                    nodeX(node),
                    nodeY(node),
                    260
                )
            ) > 0
        ) {
            return true;
        }

        return false;
    }

    function nodeHasEconomicConnection(node) {
        let homeDistance = 0;

        if (node == "" || !nodeExists(node)) {
            return false;
        }

        homeDistance = distanceBetween(
            baseX(),
            baseY(),
            nodeX(node),
            nodeY(node)
        );

        expansionConnectionRadius = max(
            280,
            mapDiagonal * 0.18
        );

        if (buildingCount("Whisper Node") <= 0) {
            return homeDistance <= mapDiagonal * 0.38;
        }

        for (outpost in myBuildingsOfType("Whisper Node")) {
            if (
                buildingExists(outpost) &&
                !buildingUnderConstruction(outpost) &&
                distanceBetween(
                    buildingX(outpost),
                    buildingY(outpost),
                    nodeX(node),
                    nodeY(node)
                ) <= expansionConnectionRadius
            ) {
                return true;
            }
        }

        return homeDistance <= mapDiagonal * 0.32;
    }

    function nodeAreaSecured(node) {

        let enemies = 0;
        let buildings = 0;

        if (node == "" || !nodeExists(node)) {

        guards = fightersNearPoint(
            nodeX(node),
            nodeY(node),
            170
        );
        enemies = count(
            enemyUnitsNear(
                nodeX(node),
                nodeY(node),
                220
            )
        );
        buildings = count(
            enemyBuildingsNear(
                nodeX(node),
                nodeY(node),
                280
            )
        );

        return (
            guards >= expansionEscortRequired &&
            enemies == 0 &&
            buildings == 0 &&
            armyHealthScore >= 0.72
        );
    }

    function expansionHardSafe(node) {
        let homeDistance = 0;
        let enemyDistance = 0;

        if (node == "" || !nodeExists(node)) {
            return false;
        }

        if (nodeCovered(node) || nodeRemaining(node) <= 250) {
            return false;
        }

        if (
            buildingNear(
                "Whisper Node",
                nodeX(node),
                nodeY(node),
                expansionCoverageRadius
            )
        ) {
            return false;
        }

        if (
            pendingExpansionNode != "" &&
            gameTime() < pendingExpansionUntil
        ) {
            return false;
        }

        if (
            rejectedExpansionNode == node &&
            gameTime() < rejectedExpansionUntil
        ) {
            return false;
        }

        if (nodeHasEnemyPresence(node)) {
            return false;
        }

        homeDistance = distanceBetween(
            baseX(),
            baseY(),
            nodeX(node),
            nodeY(node)
        );
        enemyDistance = distanceBetween(
            enemyBaseX(),
            enemyBaseY(),
            nodeX(node),
            nodeY(node)
        );

        if (enemyBaseDistance < 120) {
            return (
                nodeHasEconomicConnection(node) &&
                homeDistance <= mapDiagonal * 0.42
            );
        }

        if (enemyDistance < 230) {
            return false;
        }

        if (nodeOnFriendlySide(node)) {
            return (
                nodeHasEconomicConnection(node) ||
                homeDistance <= mapDiagonal * 0.42
            );
        }

        if (homeDistance >= enemyDistance) {
            return false;
        }

        return (
            nodeHasEconomicConnection(node) &&
            nodeAreaSecured(node) &&
            pressureMode >= 1
        );
    }

    function pendingExpansionActive() {
        let foundOutpost = false;
        let progress = 0;

        if (pendingExpansionNode == "") {
            return false;
        }

        if (!nodeExists(pendingExpansionNode)) {
            pendingExpansionNode = "";
            return false;
        }

        if (nodeCovered(pendingExpansionNode)) {
            pendingExpansionNode = "";
            pendingExpansionUntil = 0;
            pendingExpansionFailures = 0;
            return false;
        }

        foundOutpost = false;

        for (outpost in myBuildingsOfType("Whisper Node")) {
            if (
                buildingExists(outpost) &&
                distanceBetween(
                    buildingX(outpost),
                    buildingY(outpost),
                    pendingExpansionX,
                    pendingExpansionY
                ) <= expansionCoverageRadius
            ) {
                foundOutpost = true;
                progress = buildingProgress(outpost);

                if (
                    !buildingUnderConstruction(outpost) ||
                    progress >= 0.99
                ) {
                    pendingExpansionNode = "";
                    pendingExpansionUntil = 0;
                    pendingExpansionFailures = 0;
                    pendingExpansionLastProgress = 0;
                    return false;
                }

                if (
                    progress >
                    pendingExpansionLastProgress + 0.01
                ) {
                    pendingExpansionLastProgress = progress;
                    pendingExpansionProgressTime = gameTime();
                    pendingExpansionUntil = gameTime() + 35;
                }
            }
        }

        if (foundOutpost) {
            if (
                gameTime() - pendingExpansionProgressTime <= 30
            ) {
                return true;
            }

            rejectedExpansionNode = pendingExpansionNode;
            rejectedExpansionUntil = gameTime() + 25;
            pendingExpansionNode = "";
            pendingExpansionUntil = 0;
            return false;
        }

        if (gameTime() < pendingExpansionUntil) {
            return true;
        }

        pendingExpansionFailures =
            pendingExpansionFailures + 1;
        rejectedExpansionNode = pendingExpansionNode;
        rejectedExpansionUntil = gameTime() + 30;
        pendingExpansionNode = "";
        pendingExpansionUntil = 0;
        return false;
    }

    function registerPendingExpansion(node) {
        pendingExpansionNode = node;
        pendingExpansionX = nodeX(node);
        pendingExpansionY = nodeY(node);
        pendingExpansionStarted = gameTime();
        pendingExpansionUntil = gameTime() + 55;
        pendingExpansionGuardUntil = gameTime() + 35;
        pendingExpansionProgress = 0;
        pendingExpansionLastProgress = 0;
        pendingExpansionProgressTime = gameTime();
    }

    function guardPendingExpansion() {
        let enemies = 0;
        let guards = 0;

        if (!pendingExpansionActive()) {
            return false;
        }

        enemies = count(
            enemyUnitsNear(
                pendingExpansionX,
                pendingExpansionY,
                190
            )
        );
        guards = fightersNearPoint(
            pendingExpansionX,
            pendingExpansionY,
            155
        );

        if (enemies > 0) {
            guardedDefendUnits(
                myUnitsOfTypes(
                    "Lancer",
                    "Ruptor",
                    "Behemoth",
                    "Seraph"
                ),
                pendingExpansionX,
                pendingExpansionY
            );
            return true;
        }

        if (
            guards < 3 &&
            gameTime() < pendingExpansionGuardUntil &&
            !emergencyActive
        ) {
            guardedDefendUnits(
                myUnitsOfTypes("Lancer", "Ruptor"),
                pendingExpansionX,
                pendingExpansionY
            );
            return true;
        }

        return true;
    }

    function chooseSafeExpansion() {
        let best = "";
        let bestScore = 99999;
        let score = 0;
        let travel = 0;
        let richnessBonus = 0;

        if (gameTime() < expansionTick) {
            return expansionCandidate;
        }

        for (node in knownResourceNodes()) {
            if (
                nodeExists(node) &&
                !nodeCovered(node) &&
                nodeRemaining(node) > 250 &&
                expansionHardSafe(node)
            ) {
                travel = distanceBetween(
                    baseX(),
                    baseY(),
                    nodeX(node),
                    nodeY(node)
                );
                richnessBonus = min(nodeRemaining(node), 1500) / 20;
                score = nodeThreatValue(node) + travel / 8 - richnessBonus;

                if (score < bestScore) {
                    bestScore = score;
                    best = node;
                }
            }
        }

        expansionCandidate = best;
        expansionDanger = bestScore;

        if (best != "") {
            expansionCandidateX = nodeX(best);
            expansionCandidateY = nodeY(best);
        }

        expansionTick = gameTime() + 8;
        return best;
    }

    function measureCurrentEconomyCapacity() {
        coveredCrystalTotal = 0;
        productiveCoveredNodes = 0;

        for (node in knownResourceNodes()) {
            if (
                nodeExists(node) &&
                nodeCovered(node) &&
                nodeRemaining(node) > 0
            ) {
                coveredCrystalTotal =
                    coveredCrystalTotal +
                    nodeRemaining(node);
                productiveCoveredNodes =
                    productiveCoveredNodes + 1;
            }
        }

        if (productiveCoveredNodes > 0) {
            averageCoveredCrystals =
                coveredCrystalTotal /
                productiveCoveredNodes;
        } else {
            averageCoveredCrystals = 0;
        }
    }

    function updateEconomicRunway() {
        measureCurrentEconomyCapacity();

        crystalRunway = coveredCrystalTotal;
        economicCrisis = (
            productiveCoveredNodes <= 0 ||
            crystalRunway < 280
        );
        economicWarning = (
            economicCrisis ||
            crystalRunway < 850 ||
            averageCoveredCrystals < 420
        );

    }

    function expansionUtility(node) {
        let value = 0;
        let distanceHome = 0;
        let candidateRichness = 0;
        let outposts = 0;

        if (!expansionHardSafe(node)) {
            return -9999;
        }

        updateEconomicRunway();
        value = 0;
        outposts = buildingCount("Whisper Node");
        candidateRichness = nodeRemaining(node);
        distanceHome = distanceBetween(
            baseX(),
            baseY(),
            nodeX(node),
            nodeY(node)
        );

        if (productiveCoveredNodes <= 0) {
            value = value + 120;
        }

        if (economicCrisis) {
            value = value + 140;
        } else {
            if (economicWarning) {
                value = value + 65;
            }
        }

        if (incomeRate < 2) {
            value = value + 75;
        } else {
            if (incomeRate < 4) {
                value = value + 45;
            } else {
                if (incomeRate > 8) {
                    value = value - 45;
                }
            }
        }

        if (averageCoveredCrystals < 300) {
            value = value + 70;
        } else {
            if (averageCoveredCrystals < 650) {
                value = value + 35;
            } else {
                if (averageCoveredCrystals > 1300) {
                    value = value - 35;
                }
            }
        }

        value = value + min(45, candidateRichness / 35);
        value = value - distanceHome / 18;
        value = value - outposts * 11;

        if (resources() > reserveResources + 260) {
            value = value + 30;
        }

        if (unitCount("Shaper") >= desiredShapers()) {
            value = value + 15;
        } else {
            value = value - 20;
        }

        if (
            mainArmyCount() >= requiredArmy &&
            armyHealthScore >= 0.72
        ) {
            value = value + 25;
        } else {
            value = value - 35;
        }

        if (recentLosses >= 3) {
            value = value - recentLosses * 8;
        }

        if (phase == 2 || phase == 4) {
            value = value - 28;
        }

        if (phase == 3) {
            value = value - 55;
        }

        if (enemyPressure > 0 || emergencyActive) {
            value = value - 200;
        }

        if (nodeOnFriendlySide(node)) {
            value = value + 20;
        }

        return value;
    }

    function shouldExpandNow(node) {
        if (
            expansionDecisionNode == node &&
            gameTime() < expansionDecisionTick
        ) {
            return expansionDecisionApproved;
        }

        expansionDecisionNode = node;
        expansionDecisionValue = expansionUtility(node);
        expansionDecisionApproved =
            expansionDecisionValue >=
            expansionDecisionThreshold;
        expansionDecisionTick = gameTime() + 5;

        return expansionDecisionApproved;
    }

    function updateExpansionSavings() {
        let candidate = "";

        if (gameTime() < expansionOpportunityTick) {
            return expansionSavingsActive;
        }

        expansionOpportunityTick = gameTime() + 5;

        if (
            emergencyActive ||
            commandCenterUnderAttack() ||
            unitCount("Shaper") < 4 ||
            (
                mainArmyCount() < 10 &&
                !economicCrisis
            ) ||
            buildingCount("Whisper Node") >= 8 ||
            pendingExpansionActive()
        ) {
            expansionSavingsActive = false;
            expansionSavingsNode = "";
            return false;
        }

        expansionTick = gameTime();
        candidate = chooseSafeExpansion();

        if (candidate == "") {
            expansionSavingsActive = false;
            expansionSavingsNode = "";
            return false;
        }

        if (!shouldExpandNow(candidate)) {
            expansionSavingsActive = false;
            expansionSavingsNode = "";
            return false;
        }

        expansionSavingsActive = true;
        expansionSavingsNode = candidate;
        expansionSavingsTarget = max(320, reserveResources);
        expansionSavingsUntil = gameTime() + 30;
        expansionOpportunityCount =
            expansionOpportunityCount + 1;

        return true;
    }

    function shouldSaveForExpansion() {
        updateEconomicRunway();
        updateExpansionSavings();

        if (!expansionSavingsActive) {
            if (
                economicWarning &&
                !emergencyActive &&
                resources() < 320
            ) {
                return true;
            }

            return false;
        }

        if (resources() >= expansionSavingsTarget) {
            return false;
        }

        if (emergencyActive) {
            return false;
        }

        return true;
    }

    function manageSafeExpansion() {
        let candidate = "";
        let outposts = 0;

        if (!safeToExpand) {
            return false;
        }

        if (pendingExpansionActive()) {
            return false;
        }

        if (gameTime() < stalledExpansionUntil) {
            return false;
        }

        if (gameTime() - lastExpansionAttempt < 10) {
            return false;
        }

        candidate = chooseSafeExpansion();

        if (candidate == "") {
            return false;
        }

        if (
            expansionDanger > 420 ||
            !expansionHardSafe(candidate) ||
            !shouldExpandNow(candidate)
        ) {
            rejectedExpansionNode = candidate;
            rejectedExpansionUntil = gameTime() + 18;
            return false;
        }

        outposts = buildingCount("Whisper Node");
        lastExpansionCount = outposts;
        lastExpansionAttempt = gameTime();

        if (safeExpandToNode(candidate)) {
            registerPendingExpansion(candidate);
            expansionSavingsActive = false;
            expansionSavingsNode = "";
            stalledExpansionUntil = gameTime() + 55;
            return true;
        }

        stalledExpansionUntil = gameTime() + 5;
        return false;
    }

    function focusFireTarget(target) {
        let ordered = 0;
        let range = 0;
        let distance = 0;

        if (target == "" || !unitExists(target)) {
            return 0;
        }

        for (id in mainFighters()) {
            if (unitExists(id)) {
                range = unitAttackRange(id);
                distance = distanceBetween(
                    unitX(id),
                    unitY(id),
                    unitX(target),
                    unitY(target)
                );

                if (distance <= range + 34) {
                    if (guardedAttackUnit(id, target)) {
                        ordered = ordered + 1;
                    }
                }
            }
        }

        focusOrders = focusOrders + ordered;
        return ordered;
    }

    function rotateDamagedUnits() {
        let ordered = 0;
        let target = "";
        ordered = 0;
        for (id in damagedUnits(0.46)) {
            if (unitExists(id) && unitDomain(id) != "building" && !workerEscaping(id)) {
                target = nearestEnemyUnitTo(unitX(id),unitY(id),160);
                if (unitHealthRatio(id) < retreatThreshold ||
                    (target != "" && unitAttack(target) >= unitHp(id))) {
                    if (requestLocalEscape(id,2)) { ordered = ordered + 1; }
                }
            }
        }
        retreatOrders = retreatOrders + ordered;
        return ordered;
    }

    function useReadySkillsForUnit(id) {
        let used = 0;

        if (!unitExists(id)) {
            return 0;
        }

        for (skill in unitSkills(id)) {
            if (unitSkillReady(id, skill)) {
                if (useUnitSkill(id, skill)) {
                    used = used + 1;
                }
            }
        }

        return used;
    }

    function manageCombatSkills() {
        let used = 0;
        let nearbyEnemy = "";
        let damagedNearby = 0;
        let threatened = false;

        if (gameTime() < skillTick) {
            return 0;
        }

        for (id in myUnitsOfTypes(
            "Lancer",
            "Behemoth",
            "Ruptor",
            "Seer",
            "Worldspine",
            "Seraph",
            "Tempest"
        )) {
            if (unitExists(id)) {
                nearbyEnemy = nearestEnemyUnitTo(
                    unitX(id),
                    unitY(id),
                    max(80, unitAttackRange(id) + 35)
                );

                if (
                    nearbyEnemy != "" ||
                    unitHealthRatio(id) < 0.42
                ) {
                    used = used + useReadySkillsForUnit(id);
                }
            }
        }

        for (id in myUnitsOfType("Mender")) {
            if (unitExists(id) && unitSkillReady(id, "renewalWave")) {
                damagedNearby = 0;
                for (ally in unitsNear(unitX(id), unitY(id), 175)) {
                    if (unitExists(ally) && unitHealthRatio(ally) < 0.78) {
                        damagedNearby = damagedNearby + 1;
                    }
                }
                if (damagedNearby >= 2) {
                    if (useUnitSkill(id, "renewalWave")) {
                        used = used + 1;
                    }
                }
            }
        }

        for (id in myUnitsOfType("Shaper")) {
            if (unitExists(id) && unitSkillReady(id, "mendingSpores")) {
                damagedNearby = 0;
                for (ally in unitsNear(unitX(id), unitY(id), 120)) {
                    if (unitExists(ally) && unitHealthRatio(ally) < 0.72) {
                        damagedNearby = damagedNearby + 1;
                    }
                }
                for (structure in myBuildingsNear(unitX(id), unitY(id), 120)) {
                    if (
                        buildingExists(structure) &&
                        buildingHealthRatio(structure) < 0.82
                    ) {
                        damagedNearby = damagedNearby + 2;
                    }
                }
                if (damagedNearby >= 2) {
                    if (useUnitSkill(id, "mendingSpores")) {
                        used = used + 1;
                    }
                }
            }
        }

        for (id in myUnitsOfType("Wisp")) {
            if (unitExists(id) && unitSkillReady(id, "phaseSkim")) {
                threatened = nearestEnemyUnitTo(unitX(id), unitY(id), 115) != "";
                if (
                    threatened ||
                    brainGoal == 4 ||
                    distanceBetween(unitX(id), unitY(id), baseX(), baseY()) > 300
                ) {
                    if (useUnitSkill(id, "phaseSkim")) {
                        used = used + 1;
                    }
                }
            }
        }

        for (id in myUnitsOfTypes("Riptide", "Abyss Crown", "Shadefin")) {
            if (unitExists(id)) {
                nearbyEnemy = nearestEnemyUnitTo(
                    unitX(id),
                    unitY(id),
                    max(105, unitAttackRange(id) + 45)
                );
                if (nearbyEnemy != "" || unitHealthRatio(id) < 0.55) {
                    used = used + useReadySkillsForUnit(id);
                }
            }
        }

        abilityUses = abilityUses + used;
        skillTick = gameTime() + 1;
        return used;
    }

    function calculateFormationAnchor() {
        if (phase == 4 && !emergencyActive) {
            formationX = stagingX;
            formationY = stagingY;
            formationFacingX = stagingX;
            formationFacingY = stagingY;
            return;
        }
        let anchor = "";
        let targetX = 0;
        let targetY = 0;

        targetX = baseX();
        targetY = baseY();

        if (phase == 2 || phase == 4) {
            targetX = enemyBaseX();
            targetY = enemyBaseY();
        }

        if (emergencyActive) {
            targetX = defenseX;
            targetY = defenseY;
        }

        anchor = closestUnitTo(mainFighters(), targetX, targetY);

        if (anchor != "" && unitExists(anchor)) {
            formationX = unitX(anchor);
            formationY = unitY(anchor);
        } else {
            formationX = baseX();
            formationY = baseY();
        }

        formationFacingX = targetX;
        formationFacingY = targetY;
    }

    function builderCargoSafety() {
        let enemy = "";
        let distanceHome = 0;

        if (gameTime() < builderSafetyTick) {
            return;
        }

        for (id in myUnitsOfType("Shaper")) {
            if (unitExists(id) && !workerEscaping(id)) {
                enemy = nearestEnemyUnitTo(unitX(id), unitY(id), 100);

                if (enemy != "") {
                    if (
                        unitCargo(id) > 0 ||
                        unitHealthRatio(id) < 0.72
                    ) {
                        localRetreatUnit(id);
                    } else {
                        guardedMoveUnit(id, baseX(), baseY());
                    }
                } else {
                    distanceHome = distanceBetween(
                        unitX(id),
                        unitY(id),
                        baseX(),
                        baseY()
                    );

                    if (
                        unitIsIdle(id) &&
                        unitCargo(id) <= 0 &&
                        distanceHome > mapDiagonal * 0.35
                    ) {
                        guardedMoveUnit(id, economyGuardX, economyGuardY);
                    }
                }
            }
        }

        builderSafetyTick = gameTime() + 2;
    }

    function controlledDisengage() {
        phase = 3;
        cleanupMode = false;
        finishing = false;
        groupingReadySince = -1;
        regroupUntil = max(regroupUntil,gameTime()+16);
        beginSharedRegroup();
        regroupMainArmy();
    }

    function desiredMinersForNode(node) {
        let remaining = 0;
        let desired = 0;

        if (node == "" || !nodeExists(node)) {
            return 0;
        }

        remaining = nodeRemaining(node);
        desired = 3;

        if (remaining < 180) {
            desired = 1;
        } else {
            if (remaining < 420) {
                desired = 2;
            } else {
                if (remaining > 1000 && incomeRate < 5) {
                    desired = 4;
                }
            }
        }

        if (enemyPressure > 0) {
            desired = min(desired, 2);
        }

        return desired;
    }

    function coveredNodeSafeToMine(node) {
        let nearbyEnemies = 0;
        let nearbyEnemyBuildings = 0;

        if (
            node == "" ||
            !nodeExists(node) ||
            !nodeCovered(node)
        ) {
            return false;
        }

        nearbyEnemies = count(
            enemyUnitsNear(
                nodeX(node),
                nodeY(node),
                150
            )
        );
        nearbyEnemyBuildings = count(
            enemyBuildingsNear(
                nodeX(node),
                nodeY(node),
                190
            )
        );

        return (
            nearbyEnemies <= 0 &&
            nearbyEnemyBuildings <= 0
        );
    }

    function evacuateUnsafeMine(node) {
        let ordered = 0;

        if (node == "" || !nodeExists(node)) {
            return 0;
        }

        unsafeOutpostX = nodeX(node);
        unsafeOutpostY = nodeY(node);
        unsafeOutpostUntil = gameTime() + 15;

        for (id in myUnitsOfType("Shaper")) {
            if (
                unitExists(id) && !workerEscaping(id) &&
                inRange(
                    unitX(id),
                    unitY(id),
                    nodeX(node),
                    nodeY(node),
                    150
                )
            ) {
                if (localRetreatUnit(id)) {
                    ordered = ordered + 1;
                }
            }
        }

        return ordered;
    }

    function findNearestDropoffForWorker(worker) {
        let bestDistance = 0;
        let distance = 0;

        nearestDropoffX = baseX();
        nearestDropoffY = baseY();
        bestDistance = distanceBetween(
            unitX(worker),
            unitY(worker),
            baseX(),
            baseY()
        );

        for (outpost in myBuildingsOfType("Whisper Node")) {
            if (
                buildingExists(outpost) &&
                !buildingUnderConstruction(outpost)
            ) {
                distance = distanceBetween(
                    unitX(worker),
                    unitY(worker),
                    buildingX(outpost),
                    buildingY(outpost)
                );

                if (distance < bestDistance) {
                    bestDistance = distance;
                    nearestDropoffX = buildingX(outpost);
                    nearestDropoffY = buildingY(outpost);
                }
            }
        }

        return bestDistance;
    }

    function protectCargoReturns() {
        let nearestEnemy = "";

        if (gameTime() < cargoReturnTick) {
            return loadedShaperCount;
        }

        loadedShaperCount = 0;
        emptyIdleShaperCount = 0;

        for (worker in myUnitsOfType("Shaper")) {
            if (unitExists(worker) && !workerEscaping(worker)) {
                if (unitCargo(worker) > 0) {
                    loadedShaperCount =
                        loadedShaperCount + 1;

                    nearestEnemy = nearestEnemyUnitTo(
                        unitX(worker),
                        unitY(worker),
                        95
                    );

                    if (nearestEnemy != "") {
                        if (localRetreatUnit(worker)) {
                            cargoRescueOrders =
                                cargoRescueOrders + 1;
                        }
                    } else {
                        if (unitIsIdle(worker)) {
                            findNearestDropoffForWorker(worker);

                            if (
                                guardedMoveUnit(
                                    worker,
                                    nearestDropoffX,
                                    nearestDropoffY
                                )
                            ) {
                                cargoRescueOrders =
                                    cargoRescueOrders + 1;
                            }
                        }
                    }
                } else {
                    if (unitIsIdle(worker)) {
                        emptyIdleShaperCount =
                            emptyIdleShaperCount + 1;
                    }
                }
            }
        }

        cargoReturnTick = gameTime() + 1;
        return loadedShaperCount;
    }

    function rebalanceCoveredMines() {
        let desired = 0;
        let assigned = 0;
        let shortage = 0;

        if (gameTime() < miningManagerTick) {
            return false;
        }

        protectCargoReturns();
        shortage = 0;

        for (node in knownResourceNodes()) {
            if (
                nodeExists(node) &&
                nodeCovered(node) &&
                nodeRemaining(node) > 0
            ) {
                if (!coveredNodeSafeToMine(node)) {
                    evacuateUnsafeMine(node);
                } else {
                desired = desiredMinersForNode(node);
                assigned = minersAtNode(node);

                if (assigned < desired) {
                    shortage = shortage + desired - assigned;
                    if (mineNode(node, desired)) {
                        miningOrdersIssued =
                            miningOrdersIssued + 1;
                    }
                }
                }
            }
        }

        if (
            knownMineCount <= 0 &&
            loadedShaperCount <= 0
        ) {
            gatherResources();
        }

        miningManagerTick = gameTime() + 8;
        return shortage > 0;
    }

    function buildingStrategicValue(id) {
        let type = "";
        let value = 0;

        if (!buildingExists(id)) {
            return -1;
        }

        type = buildingType(id);
        value = 10;

        if (type == "Heartspire") {
            value = 200;
        }

        if (type == "Titan Crucible") {
            value = 100;
        }

        if (type == "Spawning Vault") {
            value = 80;

    function selectCleanupBuilding() {
        let best = "";
        let bestScore = -99999;
        let score = 0;
        let distance = 0;
        let defenders = 0;

        for (building in visibleEnemyBuildings()) {
            if (buildingExists(building)) {
                distance = distanceBetween(
                    formationX,
                    formationY,
                    buildingX(building),
                    buildingY(building)
                );
                defenders = count(
                    enemyUnitsNear(
                        buildingX(building),
                        buildingY(building),
                        125
                    )
                );
                score = buildingStrategicValue(building) * 4;
                score = score - distance / 3;
                score = score - defenders * 22;

                if (buildingHealthRatio(building) < 0.35) {
                    score = score + 100;
                }

                if (score > bestScore) {
                    bestScore = score;
                    best = building;
                }
            }
        }

        if (pressureVisible && pressureOpen && pressureObjective!="" && buildingExists(pressureObjective)) { return pressureObjective; }
        return best;
    }

    function cleanupTargetValid() {

        return true;

    function cleanupBlocker() {
        let blocker = "";

        blocker = nearestBlockerToArmy(105);

        if (blockerMustBeFought(blocker)) {

        return "";
    }

    function demolishCleanupTarget() {
        let blocker = "";

        if (!cleanupTargetValid()) {
            if (!acquireCleanupTarget()) {
                return false;
            }
        }

        // Checked before the blocker check below, same reasoning as
        // objectiveDecision(): a losing fight needs to actually disengage, not
        // get stuck endlessly re-triggering "focus this blocker and hold"
        // because a blocker is always in range once the armies have made
        // contact — that pattern is exactly what used to leave a cleanup this
        // close to winning stuck skirmishing forever instead of finishing.
        if (
            armyHealthScore < 0.40 ||
            armyFightRetreat
        ) {
            controlledDisengage();
            cleanupTarget = "";
            return true;
        }

        blocker = cleanupBlocker();

        if (blocker != "" && localPowerRatio < effectiveCommitThreshold()) {
            focusFireTarget(blocker);
            guardedDefendUnits(
                mainFighters(),
                formationX,
                formationY
            );
            return true;
        }

        if (blocker != "") {
            focusFireTarget(blocker);
        }

        guardedAttackUnits(
            groundFighters(),
            cleanupTargetX,
            cleanupTargetY
        );
        guardedAttackUnits(
            airFighters(),
            cleanupTargetX,
            cleanupTargetY
        );
        guardedMoveUnits(
            myUnitsOfType("Mender"),
            formationX,
            formationY
        );

        if (isWaterAt(cleanupTargetX, cleanupTargetY)) {
            guardedAttackUnits(
                navalFighters(),
                cleanupTargetX,
                cleanupTargetY
            );
        }

        return true;
    }

    function searchForRemainingBuildings() {
        if (gameTime() < cleanupSearchUntil) {
            return;
        }

        if (unitCount("Wisp") > 0) {
            safeScoutMap();
        } else {
            if (count(mainFighters()) > 0) {
                guardedAttackUnits(
                    mainFighters(),
                    enemyBaseX(),
                    enemyBaseY()
                );
            }
        }

        if (navyCount() > 0) {
            commandNavy();
        }

        cleanupSearchUntil = gameTime() + 8;
    }

    function runBuildingCleanup() {
        // Every offensive route must pass the same assembly gate.
        if (phase != 2) {
            brainExecuteAttack();
            return false;
        }
        let visibleCount = 0;

        visibleCount = count(visibleEnemyBuildings());
        lastVisibleBuildingCount = visibleCount;

        if (
            visibleCount > 0 &&
            (phase == 2 || mainArmyCount() >= requiredArmy)
        ) {
            if (!cleanupMode && !battleOutcomeRecorded) {
                recordBattleOutcome(true);
                battleOutcomeRecorded = true;
            }
            cleanupMode = true;
        }

        if (!cleanupMode) {
            return false;
        }

        finishing = true;
        phase = 2;

        if (visibleCount > 0) {
            return demolishCleanupTarget();
        }

        cleanupTarget = "";
        searchForRemainingBuildings();
        return true;
    }

    // combatUnits() includes every non-Builder unit with an attack stat above
    // zero — including armed Scouts, which harass independently. Rolling a
    // Scout into the main army caused a real bug: a scout
    // meeting one enemy while out exploring alone is very often the *closest*
    // mainFighters() unit to that encounter, so scanArmyThreats()/
    // updateLocalBattleModel() (both anchor on "whichever fighter is nearest
    // the action") register it as the whole army being under threat and send
    // everyone marching to wherever the scout happened to be. combatFighters()
    // is the same list with the Scout excluded — every fighting-group function
    // below is built from this, not combatUnits() directly, so a scout's own
    // encounters stay its own problem (see protectWisps()) instead of the whole
    // army's.
    function isDedicatedScout(wanted) {
        for (scout in myUnitsOfType("scout")) {
            if (scout == wanted) { return true; }
        }
        for (scout in myUnitsOfType("Wisp")) {
            if (scout == wanted) { return true; }
        }
        return false;
    }

    function scanMineNetwork() {
        let farthestGuardDistance = 0;
        let richestUncovered = 0;

        if (gameTime() < nextMineScan) {
            return false;
        }

        knownMineCount = 0;
        coveredMineCount = 0;
        bestExpansionNode = "";
        economyGuardX = baseX();
        economyGuardY = baseY();
        farthestGuardDistance = 0;
        richestUncovered = 0;

        for (node in knownResourceNodes()) {
            if (
                nodeExists(node) &&
                nodeRemaining(node) > 0
            ) {
                knownMineCount =
                    knownMineCount + 1;

                if (nodeCovered(node)) {
                    coveredMineCount =
                        coveredMineCount + 1;

                    if (minersAtNode(node) > 0) {
                        let guardDistance = 0;

                        guardDistance =
                            distanceBetween(
                                baseX(),
                                baseY(),
                                nodeX(node),
                                nodeY(node)
                            );

                        if (
                            guardDistance >
                            farthestGuardDistance
                        ) {
                            farthestGuardDistance =
                                guardDistance;

                            economyGuardX =
                                nodeX(node);

                            economyGuardY =
                                nodeY(node);
                        }
                    }
                } else {
                    if (
                        nodeRemaining(node) >
                        richestUncovered &&
                        expansionHardSafe(node)
                    ) {
                        richestUncovered =
                            nodeRemaining(node);

                        bestExpansionNode = node;
                    }
                }
            }
        }

        nextMineScan = gameTime() + 8;
        return true;
    }

    function isPriorityRangedEnemy(id) {
        // unitClass() is the race-agnostic role ("sniper", "artillery",
        // "antiArmor" for rocket-type units, ...) — the same for every race's
        // matching unit regardless of its flavor name, and regardless of how
        // many ranged-priority unit types a given race has. unitType() by
        // contrast returns that *specific* unit's own per-race id ("seer",
        // "longshot", "optic", ...), which is exactly the kind of hardcoded,
        // race-by-race, easy-to-miss-one check this avoids. (This previously
        // checked cls=="Worldspine" -- a unit NAME, not a class -- which matched
        // nothing and silently dropped enemy artillery out of priority-target
        // scoring in threatScore() below.)
        let cls = "";
        cls = unitClass(id);

        return (
            cls == "sniper" ||
            cls == "artillery" ||
            cls == "antiArmor"
        );
    }

    function threatScore(id, distance) {
        let score = 0;

        score = (
            unitAttackRange(id) +
            unitAttackRange(id) +
            unitAttackRange(id) +
            unitAttack(id) -
            floor(distance / 4)
        );

        if (isPriorityRangedEnemy(id)) {
            score = score + 250;
        }

        return score;
    }

    // Nudges the running "which way does trouble come from" direction estimate
    // toward wherever an actual attacker was just seen, relative to home base —
    // call this anywhere a threat scan identifies a real attacker. An
    // exponential moving average rather than a flat average so the estimate
    // keeps tracking the enemy's actual approach lanes even if earlier fights
    // happened to come from a misleading direction (e.g. a raiding party
    // looping around).
    function noteEnemyDirection(x, y) {
        let dx = 0;
        let dy = 0;
        let length = 0;

        dx = x - baseX();
        dy = y - baseY();
        length = distanceBetween(0, 0, dx, dy);

        if (length < 1) {
            return;
        }

        dx = dx / length;
        dy = dy / length;

        if (inferredEnemyObservations <= 0) {
            inferredEnemyDirX = dx;
            inferredEnemyDirY = dy;
        } else {
            inferredEnemyDirX = inferredEnemyDirX * 0.9 + dx * 0.1;
            inferredEnemyDirY = inferredEnemyDirY * 0.9 + dy * 0.1;
        }

        inferredEnemyObservations = inferredEnemyObservations + 1;
    }

    // Best guess at where the enemy's base is before it's actually been
    // scouted, projected outward from home base along the direction attackers
    // have actually been coming from. Falls back to the real (or, pre-scout,
    // harmlessly own-base) position once nothing better is known — see
    // noteEnemyDirection() and enemyBaseFound().
    function inferredEnemyBaseX() {
        if (enemyBaseFound() || inferredEnemyObservations <= 0) {
            return enemyBaseX();
        }

        return clamp(baseX() + inferredEnemyDirX * mapDiagonal * 0.5, 0, mapWidth());
    }

    function inferredEnemyBaseY() {
        if (enemyBaseFound() || inferredEnemyObservations <= 0) {
            return enemyBaseY();
        }

        return clamp(baseY() + inferredEnemyDirY * mapDiagonal * 0.5, 0, mapHeight());
    }

    // Records one attack wave's outcome (see launchAttack()'s reset of
    // battleOutcomeRecorded, attackCollapsed()'s loss, and runBuildingCleanup()'s
    // win) into a short rolling history, most recent last, capped at 6.
    function recordBattleOutcome(won) {
        let trimmed = [];
        let i = 0;
        let outcome = 0;

        outcome = 0;

        if (won) {
            outcome = 1;
        }

        recentBattleOutcomes[count(recentBattleOutcomes)] = outcome;

        if (count(recentBattleOutcomes) > 6) {
            i = 1;
            while (i < count(recentBattleOutcomes)) {
                trimmed[count(trimmed)] = recentBattleOutcomes[i];
                i = i + 1;
            }
            recentBattleOutcomes = trimmed;
        }

        if (won) {
            battleWinStreak = battleWinStreak + 1;
            battleLossStreak = 0;
        } else {
            battleLossStreak = battleLossStreak + 1;
            battleWinStreak = 0;
        }
    }

    // Recent win rate shifted to roughly -0.5 (losing lately) .. +0.5 (winning
    // lately), 0 with no history yet.
    function battleConfidence() {
        let wins = 0;
        let i = 0;

        if (count(recentBattleOutcomes) <= 0) {
            return 0;
        }

        while (i < count(recentBattleOutcomes)) {
            wins = wins + recentBattleOutcomes[i];
            i = i + 1;
        }

        return (wins / count(recentBattleOutcomes)) - 0.5;
    }

    // combatCommitThreshold (see its declaration above) is this race's baseline
    // appetite for pushing through resistance — this is that same number
    // adjusted by how the last several fights actually went: a winning streak
    // lowers the bar (confidence earned), a losing streak raises it (caution
    // earned), without ever needing an actual advantage of zero or less.
    function effectiveCommitThreshold() {
        return clamp(combatCommitThreshold - battleConfidence() * 0.3, 0.6, 1.6);
    }

    function scanWorkerThreats() {
        let bestDangerScore = 0;
        let weakestWorkerId = "";

        workerThreat = 0;
        bestDangerScore = 0;

        if (
            rememberedWorkerAttacker != "" &&
            gameTime() < workerDefenseUntil &&
            unitExists(rememberedWorkerAttacker)
        ) {
            workerThreatX =
                unitX(rememberedWorkerAttacker);

            workerThreatY =
                unitY(rememberedWorkerAttacker);
        }

        for (worker in myUnitsOfType("Shaper")) {
            if (unitExists(worker)) {
                for (enemy in visibleEnemyUnits()) {
                    if (unitExists(enemy) && unitAttack(enemy) > 0) {
                        let enemyDistance = 0;
                        let dangerRange = 0;

                        enemyDistance =
                            distanceBetween(
                                unitX(worker),
                                unitY(worker),
                                unitX(enemy),
                                unitY(enemy)
                            );

                        dangerRange =
                            unitAttackRange(enemy) + 14;

                        if (
                            enemyDistance <=
                            dangerRange + 18
                        ) {
                            workerThreat =
                                workerThreat + 1;
                            noteEnemyDirection(unitX(enemy), unitY(enemy));

                            if (
                                threatScore(
                                    enemy,
                                    enemyDistance
                                ) > bestDangerScore
                            ) {
                                bestDangerScore =
                                    threatScore(
                                        enemy,
                                        enemyDistance
                                    );

                                rememberedWorkerAttacker =
                                    enemy;

                                workerThreatX =
                                    unitX(enemy);

                                workerThreatY =
                                    unitY(enemy);
                            }
                        }
                    }
                }
            }
        }

        weakestWorkerId = weakestUnit(
            myUnitsOfType("Shaper")
        );

        if (weakestWorkerId != "") {
            if (watchedWorker == weakestWorkerId) {
                if (
                    unitHealthRatio(weakestWorkerId) <
                    watchedWorkerHealth
                ) {
                    workerThreat = max(
                        workerThreat,
                        1
                    );

                    if (
                        rememberedWorkerAttacker == ""
                    ) {
                        workerThreatX =
                            unitX(weakestWorkerId);

                        workerThreatY =
                            unitY(weakestWorkerId);
                    }
                }
            } else {
                watchedWorker = weakestWorkerId;
            }

            watchedWorkerHealth =
                unitHealthRatio(weakestWorkerId);
        }

        if (workerThreat > 0) {
            workerDefenseUntil = gameTime() + 30;
            return true;
        }

        if (
            rememberedWorkerAttacker != "" &&
            !unitExists(rememberedWorkerAttacker)
        ) {
            rememberedWorkerAttacker = "";
            workerDefenseUntil = gameTime();
        }

        return false;
    }

            }
        }
    }

    function selectDefenseTarget() {
        emergencyNow = false;

        if (
            defenseTarget != "" &&
            !unitExists(defenseTarget)
        ) {
            defenseTarget = "";
        }

        if (commandCenterUnderAttack()) {
            emergencyNow = true;

            if (baseThreatTarget != "") {
                defenseTarget = baseThreatTarget;
                defenseX = unitX(baseThreatTarget);
                defenseY = unitY(baseThreatTarget);
            } else {
                if (
                    rememberedBaseAttacker != "" &&
                    unitExists(rememberedBaseAttacker)
                ) {
                    defenseTarget =
                        rememberedBaseAttacker;

                    defenseX =
                        unitX(rememberedBaseAttacker);

                    defenseY =
                        unitY(rememberedBaseAttacker);
                } else {
                    defenseTarget = "";
                    defenseX = baseX();
                    defenseY = baseY();
                }
            }

            defenseUntil = gameTime() + 22;
        } else {
            if (workerThreat > 0) {
                emergencyNow = true;

                defenseTarget =
                    rememberedWorkerAttacker;

                defenseX = workerThreatX;
                defenseY = workerThreatY;
                defenseUntil = workerDefenseUntil;
            } else {
                if (
                    rememberedWorkerAttacker != "" &&
                    gameTime() < workerDefenseUntil &&
                    unitExists(
                        rememberedWorkerAttacker
                    )
                ) {
                    emergencyNow = true;

                    defenseTarget =
                        rememberedWorkerAttacker;

                    defenseX = workerThreatX;
                    defenseY = workerThreatY;
                    defenseUntil = workerDefenseUntil;
                } else {
                    if (baseThreat > 0) {
                        emergencyNow = true;
                        defenseTarget = baseThreatTarget;
                        defenseX = baseThreatX;
                        defenseY = baseThreatY;

                        defenseUntil =
                            gameTime() + 22;
                    } else {
                        if (assetThreat > 0) {
                            emergencyNow = true;

                            defenseTarget =
                                assetThreatTarget;

                            defenseX = assetThreatX;
                            defenseY = assetThreatY;

                            defenseUntil =
                                gameTime() + 20;

                                defenseTarget =
                                    armyThreatTarget;

                                defenseX = armyThreatX;
                                defenseY = armyThreatY;

                                defenseUntil =
                                    gameTime() + 16;
                            }
                        }
                    }
                }
            }
        }

        emergencyActive = (
            emergencyNow ||
            gameTime() < defenseUntil
        );
    }

    function rememberScoutDefenses() {
        let next = [];
        let record = [];
        let visible = false;
        next=[];
        for (building in knownEnemyBuildings()) {
            record=[]; visible=false;
            for (seen in visibleEnemyBuildings()) {
                if (seen==building && buildingExists(seen)) {
                    visible=true;
                    if (buildingAttack(seen)>0) {
                        record=[building,buildingX(seen),buildingY(seen),buildingAttackRange(seen)];
                    }
                }
            }
            if (!visible) {
                for (old in scoutKnownDefenses) { if (old[0]==building) { record=old; } }
            }
            if (count(record)==0 && knownEnemyBuildingClass(building)=="turret") {
                // Conservative perimeter until the turret's actual range is observed.
                record=[building,knownEnemyBuildingX(building),knownEnemyBuildingY(building),300];
            }
            if (count(record)>0) { next[count(next)]=record; }
        }
        scoutKnownDefenses=next;
    }

    function scoutApproachSafe(id,x,y) {
        let distance = 0;
        let standOff = 0;
        let gx = 0;
        let gy = 0;
        let sx = 0;
        let sy = 0;
        distance=distanceBetween(unitX(id),unitY(id),x,y);
        standOff=min(distance,max(6,unitAttackRange(id)-8));
        gx=clamp(x+(unitX(id)-x)*standOff/max(1,distance),28,mapWidth()-28);
        gy=clamp(y+(unitY(id)-y)*standOff/max(1,distance),28,mapHeight()-28);
        if (!inRange(gx,gy,x,y,max(6,unitAttackRange(id)))) { return false; }
        if (!localSegmentUsable(id,gx,gy)) { return false; }
        for (fraction in [0,0.33,0.66,1]) {
            sx=unitX(id)+(gx-unitX(id))*fraction;
            sy=unitY(id)+(gy-unitY(id))*fraction;
            if (siteThreatPower(sx,sy,90)>0) { return false; }
            for (defense in scoutKnownDefenses) {
                if (inRange(sx,sy,defense[1],defense[2],defense[3]+80)) { return false; }
            }
        }
        return true;
    }

    function stopScoutHarass(state,danger) {
        if (state[1]!="") { guardedStopUnit(state[0]); }

    function protectWisps() {
        let next = [];
        let state = [];
        let visible = false;
        let targetX = 0;
        let targetY = 0;
        let targetHp = 0;
        let best = "";
        let bestKind = 0;
        let bestScore = 0;
        let score = 0;
        let distance = 0;
        let limit = 0;
        if (gameTime()<nextScoutTactics) { return; }
        nextScoutTactics=gameTime()+0.2;
        rememberScoutDefenses();
        next=[];
        for (id in myUnitsOfTypes("scout","Wisp")) {
            if (unitExists(id)) {
                state=[];
                for (old in scoutHarassStates) { if (old[0]==id) { state=old; } }
                if (count(state)==0) {
                    state=[id,"",0,unitX(id),unitY(id),0,0,unitHp(id),0,unitX(id),unitY(id),gameTime(),0];
                }
                if (unitHp(id)<state[7] || siteThreatPower(unitX(id),unitY(id),65)>0 || unitHealthRatio(id)<0.55) {
                    stopScoutHarass(state,true);
                    if (unitHp(id)<state[7]) { requestLocalEscape(id,4); }
                } else {
                    if (!localEscapeOwned(id) && state[1]!="") {
                        visible=false;
                        if (state[2]==1) {
                            for (enemy in visibleEnemyUnits()) {
                                if (enemy==state[1] && unitExists(enemy)) {
                                    visible=true; targetX=unitX(enemy); targetY=unitY(enemy); targetHp=unitHp(enemy);
                                }
                            }
                        } else {
                            for (building in visibleEnemyBuildings()) {
                                if (building==state[1] && buildingExists(building)) {
                                    visible=true; targetX=buildingX(building); targetY=buildingY(building); targetHp=buildingHp(building);
                                }
                            }
                        }
                        if (!visible || gameTime()>=state[5] ||
                            !inRange(state[3],state[4],targetX,targetY,230)) {
                            stopScoutHarass(state,false);
                        } else {
                            if (!scoutApproachSafe(id,targetX,targetY)) { stopScoutHarass(state,true); }
                            else {
                                if (!inRange(unitX(id),unitY(id),state[9],state[10],12) || targetHp<state[12]) {
                                    state[9]=unitX(id); state[10]=unitY(id); state[11]=gameTime();
                                }
                                state[12]=targetHp;
                                if (gameTime()-state[11]>5) { stopScoutHarass(state,false); }
                            }
                        }
                    }
                    if (!localEscapeOwned(id) && state[1]=="" && gameTime()>=state[6] &&
                        unitHealthRatio(id)>=0.7 && unitAttack(id)>0) {
                        best=""; bestKind=0; bestScore=-999999;
                        limit=min(230,max(unitSight(id),unitAttackRange(id)+50));
                        for (enemy in visibleEnemyUnits()) {
                            if (unitExists(enemy) && enemyIsWorker(enemy) && unitAttack(enemy)<=0) {
                                distance=distanceBetween(unitX(id),unitY(id),unitX(enemy),unitY(enemy));
                                if (distance<=limit && scoutApproachSafe(id,unitX(enemy),unitY(enemy))) {
                                    score=600-distance*0.5-unitHp(enemy)*0.1;
                                    if (score>bestScore) { best=enemy; bestKind=1; bestScore=score; }
                                }
                            }
                        }
                        for (building in visibleEnemyBuildings()) {
                            if (buildingExists(building) && buildingAttack(building)<=0 &&
                                knownEnemyBuildingClass(building)!="turret") {
                                distance=distanceBetween(unitX(id),unitY(id),buildingX(building),buildingY(building));
                                if (distance<=limit && scoutApproachSafe(id,buildingX(building),buildingY(building))) {
                                    score=130-distance*0.4+(1-buildingHealthRatio(building))*80;
                                    if (buildingUnderConstruction(building)) { score=score+110; }
                                    if (knownEnemyBuildingClass(building)=="outpost") { score=score+40; }
                                    if (score>bestScore) { best=building; bestKind=2; bestScore=score; }
                                }
                            }
                        }
                        if (best!="") {
                            state[1]=best; state[2]=bestKind; state[3]=unitX(id); state[4]=unitY(id);
                            state[5]=gameTime()+10; state[8]=0;
                            state[9]=unitX(id); state[10]=unitY(id); state[11]=gameTime();
                            if (bestKind==1) { state[12]=unitHp(best); state[5]=gameTime()+14; }
                            else { state[12]=buildingHp(best); }
                        }
                    }
                    if (!localEscapeOwned(id) && gameTime()>=state[8]) {
                        if (state[1]!="") {
                            if (state[2]==1) { guardedAttackUnit(id,state[1]); }
                            else { guardedAttackMove(id,buildingX(state[1]),buildingY(state[1])); }
                            state[8]=gameTime()+2;
                        } else {
                            scoutUnit(id);
                            state[8]=gameTime()+3;
                        }
                    }
                }
                state[7]=unitHp(id);
                next[count(next)]=state;
            }
        }
        scoutHarassStates=next;
    }

        if (gameTime() < reinforcementTick) { return false; }
        reinforcementTick = gameTime() + 0.35;

        for (id in mainFighters()) {

                unitIsIdle(id) &&
                unitTarget(id) == "" &&
                !localEscapeOwned(id) &&
                !isDefenseReserved(id) &&
                !marchingContains(raidingUnits,id) &&
                !marchingContains(marchingHeld,id)
            ) {
                if (emergencyActive) {
                    guardedDefendUnit(id,defenseX,defenseY);
                } else {
                    if (phase == 2) {
                        if (marchingAnchor != "" && unitExists(marchingAnchor) && id != marchingAnchor) {
                            if (!inRange(unitX(id),unitY(id),unitX(marchingAnchor),unitY(marchingAnchor),75)) {

                        } else {

                                targetX=fieldRallyX;
                                targetY=fieldRallyY;
                                guardedDefendUnit(id,targetX,targetY);
                            }
                        }
                    }
                }
            }
        }
        return true;
    }

    function updateObjectiveProgress() {
        let lead = "";
        let distance = 0;

        lead = closestUnitTo(mainFighters(), objectiveX, objectiveY);

        if (lead == "" || !unitExists(lead)) {
            objectiveStalled = true;
            return;

        distance = distanceBetween(
            unitX(lead),
            unitY(lead),
            objectiveX,
            objectiveY
        );

        if (distance + 12 < objectiveLastDistance) {
            objectiveLastDistance = distance;
            objectiveProgressTime = gameTime();

        } else {

                objectiveStalled = true;
            }

    function blockerMustBeFought(target) {
        let lead = "";
        let distance = 0;

        if (target == "" || !unitExists(target)) {
            return false;
        }

        lead = closestUnitTo(mainFighters(), unitX(target), unitY(target));

        if (lead == "" || !unitExists(lead)) {
            return false;
        }

        distance = distanceBetween(
            unitX(lead),
            unitY(lead),
            unitX(target),
            unitY(target)
        );

        if (distance <= unitAttackRange(target) + 45) {
            return true;
        }

        if (distance <= 95 && unitAttack(target) > 0) {
            return true;
        }

        return false;
    }

    function objectiveDecision() {
        let blocker = "";

        updateLocalBattleModel();
        calculateArmyHealth();
        updateGlobalBattleModel();
        updateObjectiveProgress();

        // Checked first, ahead of the blocker check below — a losing fight
        // needs to actually disengage, not get stuck perpetually re-triggering
        // "focus this blocker and hold" every tick because a blocker is always
        // in range once the armies have made contact.
        if (
            armyHealthScore < 0.48 ||
            armyFightRetreat ||
            recentLosses >= 6
        ) {
            decisionMode = -1;
            disengageUntil = gameTime() + 14;
            return -1;
        }

        blocker = nearestBlockerToArmy(135);

        if (blockerMustBeFought(blocker)) {
            blockerTarget = blocker;
            blockerUntil = gameTime() + 4;

            // Winning locally (by this race's own bar for "winning" — see
            // combatCommitThreshold)? Keep pushing through the blocker instead
            // of parking here — microGroup()'s per-unit combat already fights
            // anything in range while attack-moving, so advancing doesn't mean
            // ignoring it, just not stalling on it. This — not a separate
            // codepath — is what actually makes "swarm through resistance" vs.
            // "hang back until the fight clearly favors you" differ by race.
            if (localPowerRatio >= effectiveCommitThreshold()) {
                decisionMode = 1;
                return 1;
            }

            decisionMode = 2;
            return 2;
        }

        // "Stalled" (see updateObjectiveProgress) just means the lead unit
        // hasn't gotten closer lately — which is completely normal while it's
        // busy winning a fight in place. Only actually hold for that if the
        // fight *isn't* going the army's way either.
        if (
            (objectiveStalled || gameTime() >= decisionUntil) &&
            localPowerRatio < effectiveCommitThreshold()
        ) {
            decisionMode = 0;
            holdUntil = gameTime() + 6;
            return 0;
        }

        if (
            localPowerRatio < 1.0 &&
            count(visibleEnemyUnits()) > 0
        ) {
            decisionMode = 0;
            holdUntil = gameTime() + 5;
            return 0;
        }

        decisionMode = 1;
        return 1;
    }

    function executeObjectiveDecision() {
        let choice = 0;

        choice = objectiveDecision();

        if (choice == -1) {
            controlledDisengage();
            return;
        }

        if (choice == 2) {
            focusFireTarget(blockerTarget);
            guardedDefendUnits(mainFighters(), formationX, formationY);
            return;
        }

        if (choice == 0) {
            guardedDefendUnits(mainFighters(), formationX, formationY);
            return;
        }

        // Advancing through/past a blocker still worth concentrating fire on —
        // keeps the push moving instead of trading evenly with everything in reach.
        if (
            blockerTarget != "" &&
            unitExists(blockerTarget) &&
            gameTime() < blockerUntil
        ) {
            focusFireTarget(blockerTarget);
        }

        guardedAttackUnits(mainFighters(), objectiveX, objectiveY);
    }

    function commandAttackWave() {
        supportLaunchUntil = gameTime() + 12;

        // enemy base hasn't actually been found yet, instead of just marching
        // toward home base as enemyBaseX/Y() harmlessly fall back to — an
        // educated guess from real attack history beats standing still.
        setPressureDestination();
        executeObjectiveDecision();

        positionEconomyGuards();
        moveMendersWithArmy();
    }

    function beginSharedRegroup() {
        regroupX=fieldRallyX;
        regroupY=fieldRallyY;

            siteThreatPower(regroupX,regroupY,75)>0) {

    function regroupMainArmy() {
        let survivors = [];
        let supportX = 0;
        let supportY = 0;

        if (!regroupActive) { beginSharedRegroup(); }
        if (gameTime()<nextRegroupOrder) { return false; }
        nextRegroupOrder=gameTime()+0.55;
        survivors=[];

        for (id in mainFighters()) {
            if (unitExists(id) && !isDefenseReserved(id) &&
                !marchingContains(raidingUnits,id)) {
                // Only a unit that is both critically hurt and still in weapons
                // danger gets its own escape. Everybody else receives the same
                // rally, and escaped units join it automatically when released.
                if (unitHealthRatio(id)<0.28 &&
                    siteThreatPower(unitX(id),unitY(id),55)>0) {
                    if (localRetreatUnit(id)) { retreatOrders=retreatOrders+1; }
                } else {
                    if (!localEscapeOwned(id)) { survivors[count(survivors)]=id; }
                }
            }
        }

        spreadAtPoint(survivors,regroupX,regroupY);

        supportX=regroupX-fieldDirX*35;
        supportY=regroupY-fieldDirY*35;
        if (isWaterAt(supportX,supportY)) { supportX=regroupX; supportY=regroupY; }
        for (id in myUnitsOfType("Mender")) {
            if (unitExists(id) && !marchingContains(survivors,id)) {
                if (unitHealthRatio(id)<0.25 &&
                    siteThreatPower(unitX(id),unitY(id),55)>0) {
                    localRetreatUnit(id);
                } else {
                    if (!localEscapeOwned(id)) {
                        if (!inRange(unitX(id),unitY(id),supportX,supportY,28)) {
                            guardedMoveUnit(id,supportX,supportY);
                        } else {
                            guardedDefendUnit(id,supportX,supportY);
                        }
                    }
                }
            }
        }
        return true;
    }

    function commandNavy() {
        let fleet = 0;
        let enemies = 0;
        let target = "";

        fleet = navyCount();

        if (fleet <= 0) {
            return false;
        }

        if (gameTime() < navalRetreatUntil) {
            localRetreatUnits(navalFighters());
            return true;
        }

        if (navalRallyX == 0 && navalRallyY == 0) {
            navalRallyX = baseX();
            navalRallyY = baseY();
        }

        target = nearestEnemyUnitTo(
            navalRallyX,
            navalRallyY,
            220
        );
        enemies = count(
            enemyUnitsNear(
                navalRallyX,
                navalRallyY,
                220
            )
        );

        if (
            target != "" &&
            enemies <= fleet + 2
        ) {
            navalMode = 2;
            navalObjectiveX = unitX(target);
            navalObjectiveY = unitY(target);
            guardedAttackUnits(
                navalFighters(),
                navalObjectiveX,
                navalObjectiveY
            );
            navalObjectiveUntil = gameTime() + 8;
            return true;
        }

        if (enemies > fleet + 2) {
            navalMode = -1;
            navalRetreatUntil = gameTime() + 12;
            localRetreatUnits(navalFighters());
            return true;
        }

        if (fleet < 3) {
            navalMode = 0;
            guardedDefendUnits(
                navalFighters(),
                navalRallyX,
                navalRallyY
            );
            return true;
        }

        navalMode = 1;
        navalObjectiveX = enemyBaseX();
        navalObjectiveY = enemyBaseY();
        guardedAttackUnits(
            navalFighters(),
            navalObjectiveX,
            navalObjectiveY
        );
        navalObjectiveUntil = gameTime() + 12;
        return true;
    }

    // All escape/kite requests share one controller and one final order pass.
    // State: id, waypoint x/y, mode, until, last x/y, progress time,
    // moving, plan time, route danger, heading x/y, blocked x/y, blocked until,
    // holding, retry time, last hp, damage pressure until.
    // Modes: 1 short kite, 2 local retreat, 3 worker escape, 4 damage with poor vision.
    function localEscapeOwned(id) {
        for (request in localEscapeRequests) {
            if (request[0] == id) { return true; }
        }
        for (state in localEscapeStates) {
            if (state[0] == id && gameTime() < state[4]) { return true; }
        }
        return false;
    }

    function requestLocalEscape(id, mode) {
        let found = false;
        found = false;
        if (!unitExists(id)) { return false; }
        for (request in localEscapeRequests) {
            if (request[0] == id) { request[1] = max(request[1], mode); found = true; }
        }
        if (!found) { localEscapeRequests[count(localEscapeRequests)] = [id,mode]; }
        return true;
    }

    function localRetreatUnit(id) { return requestLocalEscape(id,2); }
    function localRetreatUnits(list) {

    function refreshLocalEscapeThreats() {
        localEscapeThreats = [];
        for (enemy in visibleEnemyUnits()) {
            if (unitExists(enemy) && unitAttack(enemy) > 0) {
                localEscapeThreats[count(localEscapeThreats)] =
                    [unitX(enemy),unitY(enemy),unitAttackRange(enemy),unitAttack(enemy),enemy];
            }
        }
        for (enemy in visibleEnemyBuildings()) {
            if (buildingExists(enemy) && buildingAttack(enemy) > 0) {
                localEscapeThreats[count(localEscapeThreats)] =
                    [buildingX(enemy),buildingY(enemy),buildingAttackRange(enemy),buildingAttack(enemy),""];
            }
        }
    }

    function localThreatRisk(x,y) {
        let risk = 0;
        let distance = 0;
        risk = 0;
        for (threat in localEscapeThreats) {
            distance = distanceBetween(x,y,threat[0],threat[1]);
            risk = risk + (threat[3] + 3) * max(0,threat[2] + 180 - distance) / max(1,threat[2] + 180);
            risk = risk + (threat[3] + 3) * 5 * max(0,threat[2] + 18 - distance) / max(1,threat[2] + 18);
        }
        return risk;
    }

    function localUnderThreat(x,y) {
        for (threat in localEscapeThreats) {
            if (inRange(x,y,threat[0],threat[1],threat[2] + 60)) { return true; }
        }
        return false;
    }

    function localEdgePenalty(x,y) {
        let edgeX = 0;
        let edgeY = 0;
        edgeX = min(x,mapWidth() - x);
        edgeY = min(y,mapHeight() - y);
        return max(0,85 - edgeX) * 0.3 + max(0,85 - edgeY) * 0.3 +
            min(max(0,80 - edgeX),max(0,80 - edgeY)) * 0.6;
    }

    function localSegmentUsable(id,x,y) {
        let sx = 0;
        let sy = 0;
        if (unitDomain(id) == "air") { return true; }
        for (fraction in [0.25,0.5,0.75,1]) {
            sx = unitX(id) + (x - unitX(id)) * fraction;
            sy = unitY(id) + (y - unitY(id)) * fraction;
            if (unitDomain(id) == "sea") {
                if (!isWaterAt(sx,sy)) { return false; }
            } else {
                if (isWaterAt(sx,sy)) { return false; }
            }
        }
        // Terrain/collision path queries are not exposed; rejected or stuck
        // movement is handled by trying another local waypoint.
        return true;
    }

    function localRoutePeak(id,x,y) {
        let peak = 0;
        peak = 0;
        for (fraction in [0.25,0.5,0.75,1]) {
            peak = max(peak,localThreatRisk(unitX(id) + (x - unitX(id)) * fraction,
                unitY(id) + (y - unitY(id)) * fraction));
        }
        return peak;
    }

    function nearbyRetreatCover(id) {
        let best = [];
        let bestScore = 999999;
        let score = 0;
        let distance = 0;
        let x = 0;
        let y = 0;
        let currentRisk = 0;
        best = [unitX(id),unitY(id),false];
        bestScore = 999999;
        currentRisk = localThreatRisk(unitX(id),unitY(id));
        for (ally in combatUnits()) {
            if (ally != id && unitExists(ally) && !isDedicatedScout(ally) &&
                !localEscapeOwned(ally) && unitHealthRatio(ally) >= 0.6 &&
                (unitDomain(id) == "air" || unitDomain(ally) == unitDomain(id))) {
                distance = distanceBetween(unitX(id),unitY(id),unitX(ally),unitY(ally));
                if (distance <= 280 && localThreatRisk(unitX(ally),unitY(ally)) <= currentRisk + 2) {
                    score = localThreatRisk(unitX(ally),unitY(ally)) * 3 + distance * 0.03;
                    if (score < bestScore) {
                        bestScore = score;
                        best = [unitX(ally),unitY(ally),true];
                    }
                }
            }
        }
        for (building in myBuildings()) {
            if (buildingExists(building) && !buildingUnderConstruction(building)) {
                distance = distanceBetween(unitX(id),unitY(id),buildingX(building),buildingY(building));
                if (distance <= 280) {
                    // A nearby structure is optional cover, not a command to go home.
                    x = buildingX(building) + (unitX(id) - buildingX(building)) * 56 / max(1,distance);
                    y = buildingY(building) + (unitY(id) - buildingY(building)) * 56 / max(1,distance);
                    if (localThreatRisk(x,y) <= currentRisk + 2 && localSegmentUsable(id,x,y)) {
                        score = localThreatRisk(x,y) * 3 + distance * 0.03 + 2;
                        if (score < bestScore) { bestScore = score; best = [x,y,true]; }
                    }
                }
            }
        }
        return best;
    }

    function executeLocalEscapes() {
        let memory = [];
        let nextMemory = [];
        let state = [];
        let found = false;
        let duration = 0;
        let id = "";
        let threatened = false;
        let arrived = false;
        let stuck = false;
        let replan = false;
        let routeRisk = 0;
        let cover = [];
        let directions = [];
        let lengths = [];
        let candidates = [];
        let x = 0;
        let y = 0;
        let margin = 0;
        let distance = 0;
        let score = 0;
        let peak = 0;
        let bestIndex = -1;
        let bestScore = 0;
        let index = 0;
        let issued = false;
        let moving = false;
        // The candidate search below costs roughly (directions*lengths) route/threat
        // evaluations per unit that replans this tick, multiplied again by however
        // many visible threats each evaluation scans -- during a real fight many
        // units need a new escape route on the same tick, so this can add up fast.
        // The engine's per-tick step budget (see MAX_STEPS_PER_TICK in
        // ScriptEngine.ts) was raised 10x specifically to give this room to run at
        // real quality; these caps are now a generous safety net rather than a
        // tight throttle -- high enough that they essentially never bind in normal
        // play, low enough that even an extreme battle (many units, many visible
        // enemies at once) still can't blow the whole tick's budget and crash the
        // script for that tick, which is what used to happen with no cap at all.
        let replansThisTick = 0;
        let maxReplansPerTick = 3;
        // nearbyRetreatCover(id) alone costs roughly as much as several threat
        // checks (it scans nearby allies and buildings, each via its own
        // localThreatRisk call) and runs for every unit still in escape memory,
        // not just ones that replan -- a second multiplier alongside the replan
        // search above. Same reasoning: generous now that there's real budget,
        // still bounded so an extreme battle can't blow the tick.
        let processedThisTick = 0;
        let maxProcessedPerTick = 5;
        memory = localEscapeStates;
        nextMemory = [];
        refreshLocalEscapeThreats();
        for (request in localEscapeRequests) {
            found = false;
            duration = 2;
            if (request[1] == 1) { duration = 0.65; }
            if (request[1] >= 3) { duration = 4; }
            for (old in memory) {
                if (old[0] == request[0]) {
                    if (gameTime() >= old[4]) { old[3] = request[1]; }
                    else { old[3] = max(old[3],request[1]); }
                    old[4] = max(old[4],gameTime() + duration);
                    found = true;
                }
            }
            if (!found && unitExists(request[0])) {
                memory[count(memory)] = [request[0],0,0,request[1],gameTime()+duration,
                    unitX(request[0]),unitY(request[0]),gameTime(),false,-10,0,0,0,0,0,0,false,0,
                    unitHp(request[0]),0];
            }
        }
        localEscapeRequests = [];
        // Full 16-point compass -- restored now that the engine's per-tick step
        // budget (see MAX_STEPS_PER_TICK in ScriptEngine.ts) has real headroom
        // for it, together with maxReplansPerTick/maxProcessedPerTick above.
        directions = [[1,0],[-1,0],[0,1],[0,-1],[0.707,0.707],[0.707,-0.707],
            [-0.707,0.707],[-0.707,-0.707],[0.924,0.383],[0.924,-0.383],
            [-0.924,0.383],[-0.924,-0.383],[0.383,0.924],[0.383,-0.924],
            [-0.383,0.924],[-0.383,-0.924]];
        margin = min(28,min(mapWidth(),mapHeight()) / 8);
        for (state in memory) {
            id = state[0];
            if (unitExists(id)) {
                moving = unitOrder(id) == "move" || unitOrder(id) == "moveTo";
                if (gameTime() < state[4] && processedThisTick >= maxProcessedPerTick) {
                    // Over this tick's processing budget -- carry the state
                    // forward unchanged (don't touch its current order) and
                    // reconsider it again next tick.
                    nextMemory[count(nextMemory)] = state;
                } else if (gameTime() < state[4]) {
                    processedThisTick = processedThisTick + 1;
                    if (unitHp(id) < state[18]) { state[19] = gameTime() + 1.5; }
                    state[18] = unitHp(id);
                    threatened = localUnderThreat(unitX(id),unitY(id)) ||
                        gameTime() < state[19] || state[3] == 4;
                    arrived = state[8] && inRange(unitX(id),unitY(id),state[1],state[2],12);
                    cover = nearbyRetreatCover(id);
                    if (!threatened && (!cover[2] || inRange(unitX(id),unitY(id),cover[0],cover[1],65))) {
                        // Low health alone does not send a unit across the map.
                        // Stop locally once clear; ordinary orders resume after release.
                        if (moving || (!state[16] && unitOrder(id) != "idle")) { stopUnit(id); }
                        state[8] = false;
                        state[16] = true;
                        state[7] = gameTime();
                    } else {
                        state[16] = false;
                        if (distanceBetween(unitX(id),unitY(id),state[5],state[6]) >= 3) {
                            state[5] = unitX(id);
                            state[6] = unitY(id);
                            state[7] = gameTime();
                        }
                        stuck = state[8] && !arrived && gameTime() - state[7] >= 1.4;
                        if (stuck) {
                            state[13] = state[1]; state[14] = state[2];
                            state[15] = gameTime() + 4;
                        }
                        routeRisk = 0;
                        if (state[8]) { routeRisk = localRoutePeak(id,state[1],state[2]); }
                        replan = !state[8] || arrived || stuck ||
                            (gameTime() - state[9] >= 0.8 && routeRisk > state[10] + 12);
                        if (replan && gameTime() >= state[17] && replansThisTick >= maxReplansPerTick) {
                            // Over this tick's replan budget -- keep whatever route
                            // this unit already had (or none) and retry very soon
                            // rather than paying for the full candidate search now.
                            state[17] = gameTime() + 0.05;
                        } else if (replan && gameTime() >= state[17]) {
                            replansThisTick = replansThisTick + 1;
                            candidates = [];
                            lengths = [64,112,160];
                            if (state[3] == 1) { lengths = [44,72]; }
                            for (length in lengths) {
                                for (direction in directions) {
                                    x = clamp(unitX(id) + direction[0] * length,margin,mapWidth()-margin);
                                    y = clamp(unitY(id) + direction[1] * length,margin,mapHeight()-margin);
                                    distance = distanceBetween(unitX(id),unitY(id),x,y);
                                    if (distance >= 24 && localSegmentUsable(id,x,y) &&
                                        !(gameTime() < state[15] && inRange(x,y,state[13],state[14],38))) {
                                        peak = localRoutePeak(id,x,y);
                                        score = localThreatRisk(x,y) * 2 + peak * 2.5 + localEdgePenalty(x,y);
                                        // Penalize a route that first walks deeper into firing range.
                                        score = score + max(0,peak-localThreatRisk(unitX(id),unitY(id))) * 5;
                                        score = score + distance * 0.015;
                                        if (cover[2]) { score = score + distanceBetween(x,y,cover[0],cover[1]) * 0.04; }
                                        if (!stuck && state[8] &&
                                            (x-unitX(id))*state[11] + (y-unitY(id))*state[12] < 0) {
                                            score = score + 18;
                                        }
                                        candidates[count(candidates)] = [x,y,score,peak];
                                    }
                                }
                            }
                            issued = false;
                            for (attempt in candidates) {
                                if (!issued) {
                                    bestScore = 999999999;
                                    bestIndex = -1;
                                    index = 0;
                                    for (candidate in candidates) {
                                        if (candidate[2] < bestScore) { bestScore = candidate[2]; bestIndex = index; }
                                        index = index + 1;
                                    }
                                    if (bestIndex >= 0) {
                                        x = candidates[bestIndex][0]; y = candidates[bestIndex][1];
                                        candidates[bestIndex][2] = 999999999;
                                        if (moveUnitTo(id,x,y)) {
                                            state[1] = x; state[2] = y; state[8] = true;
                                            state[5] = unitX(id); state[6] = unitY(id); state[7] = gameTime();
                                            state[9] = gameTime(); state[10] = candidates[bestIndex][3];
                                            distance = max(1,distanceBetween(unitX(id),unitY(id),x,y));
                                            state[11] = (x-unitX(id))/distance;
                                            state[12] = (y-unitY(id))/distance;
                                            state[17] = 0;
                                            issued = true;
                                        }
                                    }
                                }
                            }
                            if (!issued) {
                                // No blind retreat-to-base fallback and no move-to-self.
                                state[8] = false;
                                state[17] = gameTime() + 0.5;
                                if (moving) { stopUnit(id); }
                            }
                        } else {
                            if (state[8] && !moving) {
                                if (!moveUnitTo(id,state[1],state[2])) {
                                    state[13] = state[1]; state[14] = state[2]; state[15] = gameTime()+4;
                                    state[8] = false;
                                }
                            }
                        }
                    }
                    nextMemory[count(nextMemory)] = state;
                } else {
                    if (state[8] && moving) { stopUnit(id); }
                }
            }
        }
        localEscapeStates = nextMemory;
    }

    function escapeCombatUnit(id, target) {
        return requestLocalEscape(id,2);

    function enemyIsWorker(id) {
        let type = "";
        type = unitType(id);
        return unitClass(id) == "builder" || unitClass(id) == "worker" ||
            type == "Shaper" || type == "shaper" || type == "Fabricator" ||
            type == "fabricator" || type == "Assembler" || type == "assembler";
    }

    function refreshCombatTargets() {
        let support = 0;
        let turret = 0;
        let workerThreat = false;
        // How many of our own fighters are already shooting at this enemy right
        // now -- computed once here (O(enemies*fighters), same refresh cadence
        // combat micro already runs at) rather than per attacking unit, so every
        // unit's target choice below can cheaply favor whatever the group is
        // already converging fire onto. Individually-optimal target picks spread
        // damage across many enemies at once, each surviving longer and getting
        // more shots off; concentrating fire kills one enemy at a time and
        // removes its return damage from the fight sooner -- a real "smarter as
        // a group" win, not just a smarter individual.
        let focusCount = 0;
        if (gameTime() < combatTargetRefresh) { return; }
        combatTargetRefresh = gameTime() + 0.4;
        combatTargetFacts = [];
        for (enemy in visibleEnemyUnits()) {
            if (unitExists(enemy)) {
                support = 0;
                turret = 0;
                workerThreat = false;
                focusCount = 0;
                for (guard in visibleEnemyUnits()) {
                    if (unitExists(guard) && unitAttack(guard) > 0 && inRange(
                        unitX(enemy), unitY(enemy), unitX(guard), unitY(guard), unitAttackRange(guard) + 80)) {
                        support = support + unitAttack(guard) + 3;
                    }
                }
                for (building in visibleEnemyBuildings()) {
                    if (buildingExists(building) && buildingAttack(building) > 0 && inRange(
                        unitX(enemy), unitY(enemy), buildingX(building), buildingY(building), buildingAttackRange(building) + 50)) {
                        turret = turret + buildingAttack(building) + 8;
                    }
                }
                for (worker in myUnitsOfType("Shaper")) {
                    if (unitTarget(enemy) == worker) { workerThreat = true; }
                }
                for (ally in mainFighters()) {
                    if (unitTarget(ally) == enemy) { focusCount = focusCount + 1; }
                }
                combatTargetFacts[count(combatTargetFacts)] = [enemy,support,turret,enemyIsWorker(enemy),workerThreat,focusCount];
            }
        }
    }

    function chooseCombatTarget(id) {
        let best = "";
        let bestScore = -999999;
        let score = 0;
        let distance = 0;
        let nearbyPower = 0;
        let allowed = false;
        best = "";
        bestScore = -999999;
        nearbyPower = 0;
        for (ally in mainFighters()) {
            if (unitExists(ally) && inRange(unitX(id), unitY(id), unitX(ally), unitY(ally), 140)) {
                nearbyPower = nearbyPower + unitAttack(ally) + 3;
            }
        }
        for (fact in combatTargetFacts) {
            if (unitExists(fact[0])) {
                distance = distanceBetween(unitX(id),unitY(id),unitX(fact[0]),unitY(fact[0]));
                allowed = distance <= max(unitSight(id),unitAttackRange(id) + 35);
                // A worker behind armed cover is not a free kill.
                if (fact[3] && (fact[2] > 0 || fact[1] > nearbyPower * 0.8)) { allowed = false; }
                if (allowed) {
                    score = 40 - distance * 0.5;
                    if (fact[3]) { score = score + 140; }
                    if (unitHp(fact[0]) <= unitAttack(id) * 2) { score = score + 80; }
                    if (unitAttack(fact[0]) > 0 && distance <= unitAttackRange(fact[0]) + 20) {
                        score = score + 220;
                    }
                    if (fact[4]) { score = score + 170; }
                    if (unitAttackRange(fact[0]) > 70) { score = score + 35; }
                    if (unitTarget(id) == fact[0]) { score = score + 20; }
                    // Converge with allies already shooting this target -- see
                    // refreshCombatTargets(). Capped so it nudges ties toward
                    // grouping up rather than overriding a clearly better pick
                    // (a low-HP kill or an in-range threat still wins on their
                    // own bonuses above).
                    score = score + min(120, fact[5] * 25);
                    if (score > bestScore) { bestScore = score; best = fact[0]; }
                }
            }
        }
        return best;
    }

    function endEconomyRaid() {
        for (id in raidingUnits) {
            if (unitExists(id)) { guardedMoveUnit(id, formationX, formationY); }
        }
        raidingUnits = [];
        economyRaidTarget = "";
        nextEconomyRaid = gameTime() + 8;
    }

    function pressureExposedWorkers() {
        let best = "";
        let bestScore = -999999;
        let score = 0;
        let distance = 0;
        let candidate = "";
        let candidatesOpen = true;
        let raidPower = 0;
        let targetSupport = 999999;
        let targetTurret = 999999;
        let targetVisible = false;
        let livingRaiders = [];
        if (phase == 3 || phase == 4 || homeDanger || fullHomeDefense || armyFightRetreat || armyHealthScore < 0.65) {
            if (economyRaidTarget != "") { endEconomyRaid(); }
            return;
        }
        refreshCombatTargets();
        if (economyRaidTarget == "") {
            if (gameTime() < nextEconomyRaid || count(advanceFighters()) < 3 || localPowerRatio < 1.15) { return; }
            best = "";
            bestScore = -999999;
            for (fact in combatTargetFacts) {
                if (fact[3] && fact[2] <= 0 && fact[1] <= 0 && unitExists(fact[0])) {
                    distance = distanceBetween(formationX, formationY, unitX(fact[0]), unitY(fact[0]));
                    if (distance <= 440) {
                        score = 200 - distance - unitHp(fact[0]) * 0.1;
                        if (score > bestScore) { bestScore = score; best = fact[0]; }
                    }
                }
            }
            if (best == "") { return; }
            raidingUnits = [];
            candidatesOpen = true;
            while (candidatesOpen && count(raidingUnits) < 2) {
                candidate = "";
                bestScore = -999999;
                for (id in groundFighters()) {
                    if (unitExists(id) && !localEscapeOwned(id) && unitHealthRatio(id) >= 0.7 && !isDefenseReserved(id) && id != marchingAnchor &&
                        unitAttackRange(id) <= 80 && !marchingContains(raidingUnits,id)) {
                        distance = distanceBetween(unitX(id),unitY(id),unitX(best),unitY(best));
                        score = unitSpeed(id) * 3 - distance;
                        if (distance <= 460 && score > bestScore) { bestScore = score; candidate = id; }
                    }
                }
                if (candidate == "") { candidatesOpen = false; }
                else { raidingUnits[count(raidingUnits)] = candidate; }
            }
            if (count(raidingUnits) < 2) { raidingUnits = []; nextEconomyRaid = gameTime() + 3; return; }
            economyRaidTarget = best;
            economyRaidUntil = gameTime() + 16;
        }
        targetVisible = false;
        targetSupport = 999999;
        targetTurret = 999999;
        for (fact in combatTargetFacts) {
            if (fact[0] == economyRaidTarget) {
                targetVisible = true;
                targetSupport = fact[1];
                targetTurret = fact[2];
            }
        }
        livingRaiders = [];
        raidPower = 0;
        for (id in raidingUnits) {
            if (unitExists(id) && !localEscapeOwned(id) && unitHealthRatio(id) >= 0.55) {
                livingRaiders[count(livingRaiders)] = id;
                raidPower = raidPower + unitAttack(id) + 3;
            }
        }
        if (!targetVisible || !unitExists(economyRaidTarget) || count(livingRaiders) < 2 ||
            targetTurret > 0 || targetSupport > raidPower * 0.7 || gameTime() >= economyRaidUntil ||
            !inRange(formationX,formationY,unitX(economyRaidTarget),unitY(economyRaidTarget),480)) {
            endEconomyRaid();
            return;
        }
        for (id in raidingUnits) { guardedAttackUnit(id,economyRaidTarget); }
    }

    function microGroup(
        type,
        canKite,
        retreatRatio
    ) {
        for (id in myUnitsOfType(type)) {
            if (unitExists(id)) {
                let target = "";

                if (
                    emergencyActive && (!interceptActive || fullHomeDefense || marchingContains(homeDefenseUnits,id)) &&
                    defenseTarget != "" &&
                    unitExists(defenseTarget)
                ) {
                    target = defenseTarget;
                } else {
                    target = chooseCombatTarget(id);
                }

                if (
                    unitHealthRatio(id) <
                    retreatRatio
                ) {
                    escapeCombatUnit(id, target);
                } else {
                    if (target != "" && (phase == 2 || emergencyActive ||
                        inRange(unitX(id), unitY(id), unitX(target), unitY(target),
                            unitAttackRange(id)))) {
                        let distance = 0;
                        let ownRange = 0;
                        let targetRange = 0;
                        let escapeX = 0;
                        let escapeY = 0;
                        let leashDistance = 0;
                        let targetThreatening = false;
                        let effectiveLeash = chaseLeash;

                        distance =
                            distanceBetween(
                                unitX(id),
                                unitY(id),
                                unitX(target),
                                unitY(target)
                            );

                        ownRange =
                            unitAttackRange(id);

                        targetRange =
                            unitAttackRange(target);

                        leashDistance = distanceBetween(
                            unitX(id),
                            unitY(id),
                            formationX,
                            formationY
                        );

                        targetThreatening = (
                            distance <= targetRange + 18 ||
                            inRange(
                                unitX(target),
                                unitY(target),
                                objectiveX,
                                objectiveY,
                                90
                            )
                        );

                        // "Don't let it breathe" — with the enemy's whole visible
                        // army (not just this fight) clearly broken, chase a
                        // fleeing target much further before giving up and
                        // snapping back to formation. See updateGlobalBattleModel().
                        if (pursuitMode) {
                            effectiveLeash = chaseLeash * 3;
                        }

                        if (
                            !emergencyActive &&
                            leashDistance > effectiveLeash &&
                            !targetThreatening
                        ) {
                            guardedDefendUnit(
                                id,
                                formationX,
                                formationY
                            );
                        } else {

                            if (
                                canKite &&
                                ownRange > targetRange &&
                                distance <=
                                targetRange + 3 &&
                                kiteStep == 1 &&
                                (
                                    unitSpeed(id) >=
                                    unitSpeed(target) ||
                                    ownRange >
                                    targetRange +
                                    targetRange
                                )
                            ) {
                                requestLocalEscape(id,1);
                            } else {
                                guardedAttackUnit(id, target);
                            }
                        }
                    }
                }
            }
        }
    }

    function runCombatMicro() {
        if (phase == 4 && !emergencyActive) {
            return false;
        }
        if (gameTime() < nextCombatMicro) {
            return false;
        }

        refreshCombatTargets();
        microGroup("Lancer", false, 0.18);
        microGroup("Behemoth", false, 0.22);
        microGroup("Ruptor", true, 0.34);
        microGroup("Seer", true, 0.34);
        microGroup("Worldspine", true, 0.38);
        microGroup("Seraph", true, 0.3);
        microGroup("Tempest", true, 0.34);
        microGroup("Riptide", true, 0.28);
        microGroup("Shadefin", true, 0.32);
        microGroup("Abyss Crown", true, 0.22);

        moveMendersWithArmy();

        if (kiteStep == 0) {
            kiteStep = 1;
        } else {
            kiteStep = 0;
        }

        nextCombatMicro = gameTime() + 0.4;
        return true;
    }

    function updateCompositionPlan() {
        let army = 0;
        let enemyNearBase = 0;
        let infrastructure = 0;

        if (gameTime() < compositionTick) {
            return;
        }

        army = mainArmyCount();
        enemyNearBase = count(
            enemyUnitsNear(baseX(), baseY(), 230)
        );
        infrastructure =
            buildingCount("Spawning Vault") * 9 +
            buildingCount("Titan Crucible") * 11 +
            buildingCount("Whisper Node") * 4;

        // Was hard-ceilinged at 64 no matter how strong the economy got --
        // a real economic advantage (more outposts, more production buildings)
        // should keep translating into a bigger army instead of hitting an
        // arbitrary wall. The economic throttles right below (income, resources
        // vs. reserve) already do the real job of not overproducing past what
        // the economy can support; this ceiling only needs to stay out of their
        // way, not substitute for them.
        desiredArmyCap = clamp(
            20 + infrastructure,
            26,
            150
        );

        if (incomeRate < 3) {
            desiredArmyCap = min(desiredArmyCap, 50);
        }

        if (resources() < reserveResources) {
            desiredArmyCap = min(desiredArmyCap, max(28, army));
        }

        desiredLancer = max(5, ceil(desiredArmyCap * 0.26));
        desiredRuptor = max(3, ceil(desiredArmyCap * 0.23));
        desiredSeer = max(3, ceil(desiredArmyCap * 0.20));
        desiredMender = clamp(ceil(desiredArmyCap / 14), 1, 4);
        lancerEmergencyNeed = clamp(enemyNearBase + 3, 4, 14);
        researchReserve = reserveResources + 260;

        infantryProductionMode = 1;

        if (army >= desiredArmyCap) {
            infantryProductionMode = 0;
        }

        if (
            unitCount("Lancer") > desiredLancer + 2 &&
            !emergencyActive
        ) {
            infantryProductionMode = 2;
        }

        if (emergencyActive) {
            infantryProductionMode = 3;
        }

        compositionTick = gameTime() + 4;
    }

    function armyProductionCapped() {
        updateCompositionPlan();

        if (emergencyActive) {
            return false;
        }

        if (mainArmyCount() < desiredArmyCap) {
            return false;
        }

        if (recentLosses >= 4) {
            return false;
        }

        productionCapUntil = gameTime() + 5;
        return true;
    }

    function shouldSaveForResearch() {
        if (emergencyActive) {
            return false;
        }

        if (!hasBuilding("Memory Bloom")) {
            return false;
        }

        // Start reserving for the lab chain before the army becomes enormous.
        // Otherwise continuous unit production can spend the 260 crystals needed
        // for Aerial Engineering every time the bank approaches its cost.
        if (mainArmyCount() < 12) {
            return false;
        }

        if (
            hasResearch("advancedBallistics") &&
            hasResearch("aerialEngineering") &&
            hasResearch("supportSystems") &&
            (
                !waterAvailable ||
                hasResearch("navalEngineering")
            )
        ) {
            return false;
        }

        return resources() < researchReserve;
    }

    function desiredBaseTurrets() {
        let value = 1;
        let localThreat = count(enemyUnitsNear(baseX(), baseY(), 260));

        if (localThreat >= 2 || commandCenterUnderAttack()) {
            value = value + 1;
        }

        if (localThreat >= 7 && armyStrength() < enemyPressure * 13 + 40) {
            value = value + 1;
        }

        if (mainArmyCount() >= 18 && buildingCount("Whisper Node") >= 2) {
            value = value + 1;
        }

        if (economicWarning || resources() < reserveResources + 90) {
            value = value - 1;
        }

        return clamp(value, 0, 3);
    }

    function desiredTotalGazeTurrets() {
        let value = desiredBaseTurrets();
        let exposedTerritory = 0;

        for (outpost in myBuildingsOfType("Whisper Node")) {
            if (
                buildingExists(outpost) &&
                !buildingUnderConstruction(outpost) &&
                count(enemyUnitsNear(buildingX(outpost), buildingY(outpost), 250)) > 0
            ) {
                exposedTerritory = exposedTerritory + 1;
            }
        }

        value = value + min(2, exposedTerritory);

        if (
            !economicWarning &&
            resources() > reserveResources + 360 &&
            buildingCount("Whisper Node") >= 3
        ) {
            value = value + 1;
        }

        return clamp(value, 0, 6);
    }

    function shouldBuildGazeTurret() {

        if (economicWarning || expansionSavingsActive) {
            return commandCenterUnderAttack();
        }

        if (resources() <= reserveResources + 100) {
            return false;
        }

        return true;

    function fortifyOneOutpost() {
        if (!shouldBuildGazeTurret()) {
            return false;
        }

        for (
            outpost in
            myBuildingsOfType("Whisper Node")

            if (

                !buildingUnderConstruction(outpost) &&
                !buildingNear(
                    "Gaze",
                    buildingX(outpost),
                    buildingY(outpost),
                    95
                )

                let turretX = 0;
                let turretY = 0;

                turretX = clamp(
                    buildingX(outpost) + 70,
                    0,
                    mapWidth()

                turretY = buildingY(outpost);

                if (
                    isValidBuildSpot(
                        turretX,
                        turretY
                    ) &&
                    safeConstructAt(
                        "Gaze",
                        turretX,
                        turretY
                    )
                ) {
                    return true;
                }

                turretX = clamp(
                    buildingX(outpost) - 70,
                    0,
                    mapWidth()

                if (
                    isValidBuildSpot(
                        turretX,
                        turretY
                    ) &&
                    safeConstructAt(
                        "Gaze",
                        turretX,
                        turretY
                    )
                ) {
                    return true;
                }
            }
        }

        return false;
    }

    // A resource node the AI already relies on (covered, being mined) that's
    // currently unsafe to mine (see coveredNodeSafeToMine) gets a turret of its
    // own instead of just relying on evacuateUnsafeMine()/rebalanceCoveredMines()
    // to keep pulling workers off it — so the next time an enemy unit wanders
    // by, the node defends itself and mining can resume without the main army
    // needing to be pulled back to babysit it. Same turret budget as
    // shouldBuildGazeTurret()/desiredTotalGazeTurrets() everywhere else, so this
    // doesn't uncap total turret spending — it just reprioritizes toward
    // whichever spot is actually under threat right now.
    function fortifyThreatenedNode() {
        if (!shouldBuildGazeTurret()) {
            return false;
        }

        for (node in knownResourceNodes()) {
            if (
                nodeExists(node) &&
                nodeCovered(node) &&
                !coveredNodeSafeToMine(node) &&
                !buildingNear("Gaze", nodeX(node), nodeY(node), 140)
            ) {
                let turretX = 0;
                let turretY = 0;

                turretX = clamp(nodeX(node) + 60, 0, mapWidth());
                turretY = nodeY(node);

                if (
                    isValidBuildSpot(turretX, turretY) &&
                    safeConstructAt("Gaze", turretX, turretY)
                ) {
                    return true;
                }

                turretX = clamp(nodeX(node) - 60, 0, mapWidth());

                if (
                    isValidBuildSpot(turretX, turretY) &&
                    safeConstructAt("Gaze", turretX, turretY)
                ) {
                    return true;
                }

                turretX = nodeX(node);
                turretY = clamp(nodeY(node) + 60, 0, mapHeight());

                if (
                    isValidBuildSpot(turretX, turretY) &&
                    safeConstructAt("Gaze", turretX, turretY)
                ) {
                    return true;
                }

                turretY = clamp(nodeY(node) - 60, 0, mapHeight());

                if (
                    isValidBuildSpot(turretX, turretY) &&
                    safeConstructAt("Gaze", turretX, turretY)
                ) {
                    return true;
                }
            }
        }

        return false;
    }

    function manageConstruction() {
        if (gameTime() < nextBuildAttempt) {
            return false;
        }

        if (fortifyThreatenedNode()) {
            nextBuildAttempt = gameTime() + 10;
            return true;
        }

        // An island map (no land path to anyone — see isIsolatedByWater()) means
        // Tidewomb isn't a late-game option, it's the only way to ever reach
        // another base at all. Hold off on every other construction option
        // below (research, outposts, turrets, Titan Crucible, ...) until it's
        // built, judging affordability against the building's real cost rather
        // than the (much larger, constantly-growing) economic reserve —
        // otherwise a cheaper building's lower threshold keeps winning the race
        // for the same resources every tick this can't yet afford one outright.
        if (
            isIsolated &&
            !hasBuilding("Tidewomb")
        ) {
            if (
                gameTime() >= 40 &&
                resources() >= buildingCost("Tidewomb")
            ) {
                if (safeConstruct("Tidewomb")) {
                    nextBuildAttempt = gameTime() + 20;
                    return true;
                }
            }

            nextBuildAttempt = gameTime() + 5;
            return false;
        }

        if (
            !hasBuilding("Memory Bloom") &&
            hasBuilding("Titan Crucible") &&
            mainArmyCount() >= 9 &&
            gameTime() >= 110 &&
            !expansionSavingsActive
        ) {
            if (safeConstruct("Memory Bloom")) {
                nextBuildAttempt = gameTime() + 12;
                return true;
            }
        }

        if (
            buildingCount("Whisper Node") <
            desiredOutposts()
        ) {
            if (manageSafeExpansion()) {
                nextBuildAttempt = gameTime() + 10;
                nextMineScan = gameTime();
                return true;
            }
        }

        if (
            shouldBuildGazeTurret() &&
            baseTurretCount() < desiredBaseTurrets()
        ) {
            if (safeConstruct("Gaze")) {
                nextBuildAttempt = gameTime() + 10;
                return true;
            }
        }

        if (shouldBuildGazeTurret() && fortifyOneOutpost()) {
            nextBuildAttempt = gameTime() + 10;
            return true;
        }

        if (
            buildingCount("Spawning Vault") <
            desiredBarracks()
        ) {
            if (safeConstruct("Spawning Vault")) {
                nextBuildAttempt = gameTime() + 12;
                return true;
            }
        }

        if (
            buildingCount("Titan Crucible") <
            desiredFactories()
        ) {
            if (safeConstruct("Titan Crucible")) {
                nextBuildAttempt = gameTime() + 14;
                return true;
            }
        }

        if (
            waterAvailable &&
            !hasBuilding("Tidewomb") &&
            hasBuilding("Memory Bloom") &&
            gameTime() >= 150 &&
            resources() > reserveResources + 160
        ) {
            if (safeConstruct("Tidewomb")) {
                nextBuildAttempt = gameTime() + 25;
                return true;
            }
        }

        nextBuildAttempt = gameTime() + 7;
        return false;
    }

    function manageResearch() {
        let bestResearch = "";
        let bestScore = -9999;
        let score = 0;

        if (
            !hasBuilding("Memory Bloom") ||
            gameTime() < nextResearchAttempt
        ) {
            return false;
        }

        if (shouldSaveForExpansion()) {
            nextResearchAttempt = gameTime() + 5;
            return false;
        }

        if (!hasResearch("fieldLogistics")) {
            score =
                unitCount("Shaper") * 12 +
                buildingCount("Whisper Node") * 34 +
                productiveCoveredNodes * 22;
            if (economicWarning) {
                score = score + 80;
            }
            if (score > bestScore) {
                bestScore = score;
                bestResearch = "fieldLogistics";
            }
        }

        if (!hasResearch("infantryTactics")) {
            score =
                unitCount("Lancer") * 18 +
                unitCount("Ruptor") * 10 +
                unitCount("Seer") * 10;
            if (enemyPressure > 0) {
                score = score + 45;
            }
            if (score > bestScore) {
                bestScore = score;
                bestResearch = "infantryTactics";
            }
        }

        if (!hasResearch("compositeArmor")) {
            score =
                unitCount("Behemoth") * 28 +
                frontlineCount() * 8 +
                recentLosses * 10;
            if (commandCenterUnderAttack()) {
                score = score + 55;
            }
            if (score > bestScore) {
                bestScore = score;
                bestResearch = "compositeArmor";
            }
        }

        if (!hasResearch("advancedBallistics")) {
            score =
                unitCount("Ruptor") * 22 +
                unitCount("Seer") * 24 +
                unitCount("Worldspine") * 32;
            if (score > bestScore) {
                bestScore = score;
                bestResearch = "advancedBallistics";
            }
        }

        // Aerial Engineering requires Advanced Ballistics. It also unlocks the
        // aircraft whose count used to be its only score, creating a zero-score
        // circular dependency. Once the prerequisite is complete, make this
        // mandatory bridge in the lab chain a genuine priority.
        if (!hasResearch("aerialEngineering") &&
            hasResearch("advancedBallistics")) {
            score =
                600 +
                unitCount("Seraph") * 30 +
                unitCount("Tempest") * 34;
            if (waterAvailable) {
                score = score + 18;
            }
            if (score > bestScore) {
                bestScore = score;
                bestResearch = "aerialEngineering";
            }
        }

        if (!hasResearch("navalEngineering") &&
            hasBuilding("Tidewomb")) {
            score = -40;
            if (waterAvailable) {
                score = score + 70;
            }
            if (hasBuilding("Tidewomb")) {
                score = score + 80;
            }
            score = score + navyCount() * 32;
            if (score > bestScore) {
                bestScore = score;
                bestResearch = "navalEngineering";
            }
        }

        // Never let a locked research win the score comparison. The engine
        // correctly rejects it before Aerial Engineering, but repeatedly choosing
        // that impossible winner previously blocked the prerequisite forever.
        if (!hasResearch("supportSystems") &&
            hasResearch("aerialEngineering")) {
            score =
                unitCount("Mender") * 38 +
                unitCount("Shaper") * 8 +
                damagedArmyCount * 9;
            if (buildingCount("Whisper Node") >= 2) {
                score = score + 25;
            }
            if (score > bestScore) {
                bestScore = score;
                bestResearch = "supportSystems";
            }
        }

        if (bestResearch != "" && safeResearch(bestResearch)) {
            nextResearchAttempt = gameTime() + 12;
            return true;
        }

        nextResearchAttempt = gameTime() + 4;
        return false;
    }

    // Keep a small screen at home while an independently assembled wave attacks.
    function isDefenseReserved(id) {
        return marchingContains(standingDefenders,id) ||
            (interceptActive && marchingContains(homeDefenseUnits,id));
    }

    function advanceFighters() {
        let result = [];
        result = [];
        for (id in mainFighters()) {
            if (unitExists(id) && !isDefenseReserved(id) && !localEscapeOwned(id) &&
                !marchingContains(raidingUnits,id) && unitHealthRatio(id) >= 0.5) {
                result[count(result)] = id;
            }
        }
        return result;
    }

    function updateForwardRally() {
        let targetX = 0;
        let targetY = 0;
        let distance = 0;
        let projection = 0;
        let reach = 0;
        let x = 0;
        let y = 0;
        let found = false;
        targetX = inferredEnemyBaseX();
        targetY = inferredEnemyBaseY();
        if (pressureObjective != "") { targetX = pressureX; targetY = pressureY; }
        distance = distanceBetween(baseX(),baseY(),targetX,targetY);
        if (distance < 100) {
            targetX = mapWidth()/2; targetY = mapHeight()/2;
            distance = distanceBetween(baseX(),baseY(),targetX,targetY);
        }
        if (distance < 1) { targetX = baseX()+100; distance = 100; }
        fieldDirX = (targetX-baseX())/max(1,distance);
        fieldDirY = (targetY-baseY())/max(1,distance);
        reach = min(180,distance*0.35);
        for (building in myBuildings()) {
            if (buildingExists(building) && !buildingUnderConstruction(building) &&
                inRange(baseX(),baseY(),buildingX(building),buildingY(building),400)) {
                projection = (buildingX(building)-baseX())*fieldDirX +
                    (buildingY(building)-baseY())*fieldDirY;
                reach = max(reach,min(320,projection+65));
            }
        }
        fieldRallyX = clamp(baseX()+fieldDirX*60,28,mapWidth()-28);
        fieldRallyY = clamp(baseY()+fieldDirY*60,28,mapHeight()-28);
        found = false;
        for (factor in [1,0.8,0.6,0.4]) {
            for (side in [0,60,-60]) {
                x = clamp(baseX()+fieldDirX*reach*factor-fieldDirY*side,28,mapWidth()-28);
                y = clamp(baseY()+fieldDirY*reach*factor+fieldDirX*side,28,mapHeight()-28);
                if (!found && !isWaterAt(x,y) &&
                    !isWaterAt((x+baseX())/2,(y+baseY())/2) && siteThreatPower(x,y,70)<=0) {
                    fieldRallyX=x; fieldRallyY=y; found=true;
                }
            }
        }
    }

    function updateStandingDefense() {
        let keep = [];
        let best = "";
        let bestDistance = 0;
        let distance = 0;
        let searching = true;
        keep = [];
        if (mainArmyCount()<7) { standingDefenders=[]; return; }
        for (id in standingDefenders) {
            if (unitExists(id) && !localEscapeOwned(id) && unitHealthRatio(id)>=0.5) {
                keep[count(keep)]=id;
            }
        }
        standingDefenders=keep;
        searching=true;
        while (count(standingDefenders)<2 && searching) {
            best=""; bestDistance=999999;
            for (id in groundFighters()) {
                if (unitExists(id) && !localEscapeOwned(id) && unitHealthRatio(id)>=0.5 &&
                    !marchingContains(raidingUnits,id) && !marchingContains(standingDefenders,id)) {
                    distance=distanceBetween(unitX(id),unitY(id),fieldRallyX,fieldRallyY);
                    if (distance<bestDistance) { bestDistance=distance; best=id; }
                }
            }
            if (best=="") { searching=false; }
            else { standingDefenders[count(standingDefenders)]=best; }
        }
    }

    function spreadAtPoint(list,x,y) {
        let index = 0;
        let lane = 0;
        let row = 0;
        let columns = 0;
        let rows = 0;
        let px = 0;
        let py = 0;
        index=0;
        columns=max(1,min(4,count(list)));
        rows=max(1,ceil(count(list)/columns));
        for (id in list) {
            if (unitExists(id) && !localEscapeOwned(id)) {
                lane=index-floor(index/columns)*columns-(columns-1)/2;
                row=floor(index/columns)-(rows-1)/2;
                px=clamp(x-fieldDirY*lane*36-fieldDirX*row*30,28,mapWidth()-28);
                py=clamp(y+fieldDirX*lane*36-fieldDirY*row*30,28,mapHeight()-28);
                // A coastal formation slot can land in water even though its
                // center is valid.  Falling back to the shared center keeps that
                // unit grouped instead of silently leaving it with no order.
                if (unitDomain(id)!="air" && isWaterAt(px,py)) { px=x; py=y; }
                if (unitDomain(id)=="air" || !isWaterAt(px,py)) {
                    if (!inRange(unitX(id),unitY(id),px,py,22)) { guardedMoveUnit(id,px,py); }
                    else { guardedDefendUnit(id,px,py); }
                }
                index=index+1;
            }
        }
    }

    function positionForwardDefense() {
        if (fullHomeDefense && homeDanger) { return; }
        if (gameTime()<nextFieldOrders) { return; }
        nextFieldOrders=gameTime()+0.8;
        if (!interceptActive) {
            issuingDefenseOrders=true;
            spreadAtPoint(standingDefenders,fieldRallyX,fieldRallyY);
            issuingDefenseOrders=false;
        }
        if (phase!=2 && phase!=4 && phase!=3 && !homeDanger) {
            spreadAtPoint(advanceFighters(),fieldRallyX+fieldDirX*45,fieldRallyY+fieldDirY*45);
        }
    }

    function objectiveDefenderPower(x,y) {
        let power = 0;
        power=0;
        for (enemy in visibleEnemyUnits()) {
            if (unitExists(enemy) && unitAttack(enemy)>0 &&
                inRange(x,y,unitX(enemy),unitY(enemy),unitAttackRange(enemy)+110)) {
                power=power+unitAttack(enemy)+3;
            }
        }
        for (enemy in visibleEnemyBuildings()) {
            if (buildingExists(enemy) && buildingAttack(enemy)>0 &&
                inRange(x,y,buildingX(enemy),buildingY(enemy),buildingAttackRange(enemy)+70)) {
                power=power+buildingAttack(enemy)*3;
            }
        }
        return power;
    }

    function updatePressureObjective() {
        let best = "";
        let score = 0;
        let bestScore = -999999;
        let type = "";
        let x = 0;
        let y = 0;
        let visible = false;
        let age = 0;
        let defenders = 0;
        let power = 0;
        let guarded = false;
        let bestVisible = false;
        let bestOpen = false;
        let bestFortified = false;
        let bestX = 0;
        let bestY = 0;
        let bestAge = 0;
        if (gameTime()<nextPressureScan) { return; }
        nextPressureScan=gameTime()+1;
        best=""; bestScore=-999999; bestVisible=false; bestOpen=false; bestFortified=false;
        power=totalAttack(advanceFighters());
        // Use actual remembered positions; unseen buildings never masquerade as visible.
        for (building in knownEnemyBuildings()) {
            x=knownEnemyBuildingX(building); y=knownEnemyBuildingY(building);
            age=knownEnemyBuildingAge(building);
            visible=false;
            for (seen in visibleEnemyBuildings()) { if (seen==building) { visible=true; } }
            type=knownEnemyBuildingClass(building);
            defenders=objectiveDefenderPower(x,y);
            guarded=false;
            for (tower in knownEnemyBuildings()) {
                if (knownEnemyBuildingClass(tower)=="turret" &&
                    inRange(x,y,knownEnemyBuildingX(tower),knownEnemyBuildingY(tower),260)) { guarded=true; }
            }
            score=130-distanceBetween(fieldRallyX,fieldRallyY,x,y)*0.08-min(80,age)*0.6-defenders*2;
            if (type=="outpost") { score=score+90; }
            if (type=="barracks" || type=="factory") { score=score+75; }
            if (type=="researchLab" || type=="shipyard") { score=score+45; }
            if (type=="commandCenter") { score=score+25; }
            if (type=="turret" || guarded) { score=score-170; }
            if (visible) {
                if (buildingUnderConstruction(building)) { score=score+100; }
                score=score+(1-buildingHealthRatio(building))*80;
            }
            if (building==pressureObjective) { score=score+30; }
            if (score>bestScore) {
                bestScore=score; best=building; bestX=x; bestY=y; bestAge=age;
                bestVisible=visible;
                bestFortified=guarded || type=="turret";
                bestOpen=visible && !guarded && defenders<=power*0.4 && buildingAttack(building)<=0;
            }
        }
        pressureObjective=best;
        pressureVisible=bestVisible;
        pressureOpen=bestOpen;
        pressureFortified=bestFortified;
        if (best!="") { pressureX=bestX; pressureY=bestY; pressureAge=bestAge; }
    }

    function attackMinimum() {
        if (pressureObjective!="" && pressureOpen) { return 6; }
        if (pressureObjective!="" && !pressureVisible && !pressureFortified && pressureAge<=45) { return 8; }
        return requiredArmy;
    }

    function setPressureDestination() {
        let x = 0;
        let y = 0;
        let distance = 0;
        let lead = "";
        let standoff = 80;
        x=inferredEnemyBaseX(); y=inferredEnemyBaseY();
        if (pressureObjective!="") {
            x=pressureX; y=pressureY;
            if (!pressureVisible) {
                // Approach a scouted site, regain vision, then choose the real attack.
                lead=closestUnitTo(advanceFighters(),x,y);
                standoff=80;
                if (lead!="" && unitExists(lead)) { standoff=clamp(unitSight(lead)*0.7,40,170); }
                distance=distanceBetween(formationX,formationY,x,y);
                if (distance>standoff+10) {
                    x=x+(formationX-x)*standoff/distance;
                    y=y+(formationY-y)*standoff/distance;
                } else { x=formationX; y=formationY; }
            }
        }
        if (distanceBetween(objectiveX,objectiveY,x,y)>35 || decisionMode<=0) {
            beginObjective(x,y,1,30);
        }
    }

    function readyToAttack() {
        let fighters = [];
        let front = 0;
        let ranged = 0;
        fighters=advanceFighters(); front=0; ranged=0;
        for (id in fighters) {
            // Same class-based split frontlineCount()/rangedFighters() use
            // elsewhere -- NOT a raw attackRange number. This used to check
            // "attackRange<=45", but every unit in this race's real stat table
            // has range 120+ (soldier archetype=120, tank archetype=165, ...),
            // so front was ALWAYS 0 and this function could only ever return
            // true through the lucky "already-scouted a soft/open target"
            // shortcut above (attackMinimum()<requiredArmy) -- never through the
            // normal path. That's the actual reason the army could build up to
            // any size and never launch: readyToAttack() was silently
            // unsatisfiable by design, not by any real army-composition problem.
            let cls = unitClass(id);
            if (cls == "infantry" || cls == "armor") { front = front + 1; }
            else { ranged = ranged + 1; }
        }
        if (homeDanger && fullHomeDefense) { return false; }
        if (count(fighters)<attackMinimum()) { return false; }
        if (attackMinimum()<requiredArmy) { return true; }
        // Only a minimal frontline sanity floor, not a rigid 4/2 split -- the
        // count(fighters)>=attackMinimum() check above already guarantees a real
        // army; requiring an exact composition on top of that let a perfectly
        // attack-ready force (correct size, just composition-skewed toward
        // ranged/support that tick) sit idle indefinitely while resources piled
        // up unspent, which is exactly what was happening.
        return front>=2;
    }

    function beginGrouping() {
        phase=4; groupingStarted=gameTime(); groupingDeadline=0; groupingReadySince=-1;
        regroupActive=false;
        cleanupMode=false; finishing=false;
        groupingWave=[]; groupingSupport=[];
        for (id in advanceFighters()) {
            if (count(groupingWave)<max(24,attackMinimum())) { groupingWave[count(groupingWave)]=id; }
        }
        groupingMinimum=attackMinimum(); groupingTargetCount=count(groupingWave);
        stagingX=fieldRallyX+fieldDirX*35; stagingY=fieldRallyY+fieldDirY*35;
        stagingX=clamp(stagingX,28,mapWidth()-28); stagingY=clamp(stagingY,28,mapHeight()-28);
        for (id in myUnitsOfType("Mender")) {
            if (unitExists(id) && !localEscapeOwned(id) && unitHealthRatio(id)>=0.55 &&
                inRange(unitX(id),unitY(id),stagingX,stagingY,200)) {
                groupingSupport[count(groupingSupport)]=id;
            }
        }
        nextArmyOrder=gameTime();
        commandGrouping();
    }

    function commandGrouping() {
        let alive = [];
        if (gameTime()<nextArmyOrder) { return; }
        nextArmyOrder=gameTime()+0.6;
        alive=[];
        for (id in groupingWave) {
            if (unitExists(id) && !isDefenseReserved(id) && !localEscapeOwned(id) && unitHealthRatio(id)>=0.5) {
                alive[count(alive)]=id;
            }
        }
        groupingWave=alive;
        // Replace losses without making every newly produced unit another mandatory wait.
        for (id in advanceFighters()) {
            if (count(groupingWave)<groupingMinimum && !marchingContains(groupingWave,id)) {
                groupingWave[count(groupingWave)]=id;
            }
        }
        spreadAtPoint(groupingWave,stagingX,stagingY);
        guardedMoveUnits(groupingSupport,stagingX-fieldDirX*45,stagingY-fieldDirY*45);
    }

    function groupingReady() {
        let assembled = false;
        assembled=phase==4 && !fullHomeDefense && count(groupingWave)>=groupingMinimum;
        if (groupingMinimum<requiredArmy && attackMinimum()>groupingMinimum) { assembled=false; }
        for (id in groupingWave) {
            if (!unitExists(id) || localEscapeOwned(id) || isDefenseReserved(id) ||
                !inRange(unitX(id),unitY(id),stagingX,stagingY,110)) { assembled=false; }
        }
        for (id in groupingSupport) {
            if (unitExists(id) && unitHealthRatio(id)>=0.55 && !localEscapeOwned(id) &&
                !inRange(unitX(id),unitY(id),stagingX,stagingY,115)) { assembled=false; }
        }
        if (!assembled) { groupingReadySince=-1; return false; }
        if (groupingReadySince<0) { groupingReadySince=gameTime(); }
        return gameTime()>=groupingReadySince+3;
    }

    function launchAttack() {
        if (!groupingReady()) { return false; }
        phase = 2;
        marchingWave = [];
        for (id in groupingWave) { marchingWave[count(marchingWave)] = id; }
        for (id in groupingSupport) {
            if (unitExists(id) && unitHealthRatio(id) >= 0.45) {
                marchingWave[count(marchingWave)] = id;
            }
        }
        marchingHeld = [];
        marchingRegrouping = false;
        launchPower = totalAttack(groupingWave);
        launchCount = count(groupingWave);
        lossesAtLaunch = unitsLost();
        attackWaveMembers = groupingWave;
        battleOutcomeRecorded = false;

        requiredArmy = min(
            requiredArmy + 2,
            20
        );

        commandAttackWave();
    }

    function attackCollapsed() {
        let waveLosses = 0;
        let living = 0;
        let power = 0;
        waveLosses=0; living=0; power=0;
        for (id in attackWaveMembers) {
            if (unitExists(id)) { living=living+1; power=power+unitAttack(id); }
            else { waveLosses=waveLosses+1; }
        }
        // A wave that's actually winning its current fight (by this race's own
        // bar for "winning" -- see effectiveCommitThreshold()) must not retreat
        // just because it's down from its ORIGINAL launch strength: in a real
        // fight the enemy is losing units too, so a costly-but-winning fight
        // reads as "collapsed" by the old power/living ratios alone, and pulling
        // back from a fight you're currently winning is strictly worse than
        // finishing it. This was the actual reason attacks kept fizzling into
        // permanent stalemates -- two armies of similar strength trade evenly,
        // BOTH sides hit these same ratios at roughly the same time, and BOTH
        // retreat simultaneously before either finishes the other off. Only the
        // near-wipeout floor (living<3) stays absolute -- no amount of "winning
        // the ratio" saves a wave down to a couple of units.
        if (living<3) { return true; }
        if (localPowerRatio >= effectiveCommitThreshold()) { return false; }
        return waveLosses>=max(3,floor(launchCount/3)) ||
            power*3<launchPower*2 || living*3<launchCount*2;
    }

    function beginRecovery() {
        phase = 3;
        cleanupMode = false;
        finishing = false;
        groupingReadySince = -1;
        // Was +18: in a mirror-strength matchup both sides retreat to recover at
        // roughly the same time, and a long mandatory regroup window just hands
        // the enemy the same amount of free healing time back -- net zero, war
        // of attrition forever. Shorter window means less enemy recovery time
        // gets handed back for free too.
        regroupUntil = gameTime() + 10;
        attackDeadline = gameTime() + 70;
        beginSharedRegroup();
        regroupMainArmy();

    }

    function brainConsider(goal, score) {
        let adjusted = 0;

        adjusted = score;

        if (goal == brainGoal) {
            adjusted = adjusted + brainCurrentGoalBonus;
        }

        if (
            brainStalled &&
            goal == brainGoal
        ) {
            adjusted = adjusted - 180;
        }

        if (adjusted > brainBestScore) {
            brainBestScore = adjusted;
            brainBestGoal = goal;
        }
    }

    function brainMeasureKnowledge() {
        brainKnownUsefulNodes = 0;
        brainUncoveredNodes = 0;

        for (node in knownResourceNodes()) {
            if (
                nodeExists(node) &&
                nodeRemaining(node) > 0
            ) {
                brainKnownUsefulNodes =
                    brainKnownUsefulNodes + 1;

                if (!nodeCovered(node)) {
                    brainUncoveredNodes =
                        brainUncoveredNodes + 1;
                }
            }
        }

        brainKnowledgeNeed = 0;

        if (brainKnownUsefulNodes <= 1) {
            brainKnowledgeNeed = 100;
        } else {
            if (brainKnownUsefulNodes <= 3) {
                brainKnowledgeNeed = 55;
            }
        }

        if (economicWarning) {
            brainKnowledgeNeed =
                brainKnowledgeNeed + 65;
        }

        if (economicCrisis) {
            brainKnowledgeNeed =
                brainKnowledgeNeed + 120;
        }
    }

    function brainMeasureReadiness() {
        brainArmyReadiness = 0;

        if (mainArmyCount() >= requiredArmy) {
            brainArmyReadiness =
                brainArmyReadiness + 40;
        }

        if (frontlineCount() >= 4) {
            brainArmyReadiness =
                brainArmyReadiness + 20;
        }

        if (rangedCount() >= 3) {
            brainArmyReadiness =
                brainArmyReadiness + 20;
        }

        if (armyHealthScore >= 0.72) {
            brainArmyReadiness =
                brainArmyReadiness + 25;
        } else {
            brainArmyReadiness =
                brainArmyReadiness - 35;
        }

        if (recentLosses >= 4) {
            brainArmyReadiness =
                brainArmyReadiness -
                recentLosses * 8;
        }

        if (localPowerRatio > 1.35) {
            brainArmyReadiness =
                brainArmyReadiness + 25;
        }
    }

    function brainMeasureEconomy() {
        brainEconomyHealth = 0;

        if (incomeRate >= 6) {
            brainEconomyHealth =
                brainEconomyHealth + 60;
        } else {
            if (incomeRate >= 3) {
                brainEconomyHealth =
                    brainEconomyHealth + 30;
            } else {
                brainEconomyHealth =
                    brainEconomyHealth - 50;
            }
        }

        if (productiveCoveredNodes >= 2) {
            brainEconomyHealth =
                brainEconomyHealth + 35;
        }

        if (economicWarning) {
            brainEconomyHealth =
                brainEconomyHealth - 60;
        }

        if (economicCrisis) {
            brainEconomyHealth =
                brainEconomyHealth - 150;
        }

        if (resources() >= reserveResources) {
            brainEconomyHealth =
                brainEconomyHealth + 25;
        }
    }

    function brainObserve() {
        detectWaterMap();
        scanMineNetwork();
        scanWorkerThreats();
        scanBaseThreats();
        scanAssetThreats();
        scanArmyThreats();
        selectDefenseTarget();
        updateStrategicModes();
        updateIncomeModel();
        updateLossModel();
        calculateArmyHealth();
        updateLocalBattleModel();
        updateGlobalBattleModel();
        computeResourceReserve();
        updateEconomicRunway();
        updateCompositionPlan();
        protectCargoReturns();
        currentPower = calculateMainPower();
        enemyBaseDistance = distanceBetween(
            baseX(),
            baseY(),
            enemyBaseX(),
            enemyBaseY()
        );

        // armyThreat counts every visible enemy near your main fighters — while
        // actively pushing an attack (phase 2) that's just combat happening, not
        // a reason to score Defend higher and abort the attack. selectDefenseTarget()
        // already excludes armyThreat from triggering emergencyNow during phase 2
        // for the same reason; this keeps brainThreatLevel's baseline consistent
        // with that instead of still spiking scoreDefend off the very fight the
        // AI is supposed to be winning.
        brainThreatLevel =
            baseThreat * 3 +
            workerThreat * 2 +
            assetThreat * 2;

        if (phase != 2) {
            brainThreatLevel = brainThreatLevel + armyThreat;
        }

        brainVisibleBuildings =
            count(visibleEnemyBuildings());

        brainMeasureKnowledge();
        brainMeasureReadiness();
        brainMeasureEconomy();

        expansionTick = gameTime();
        brainCandidateNode = chooseSafeExpansion();

        if (
            resources() != brainLastResources ||
            mainArmyCount() != brainLastArmy ||
            buildingsConstructed() != brainLastBuildings ||
            brainKnownUsefulNodes != brainLastKnownNodes
        ) {
            brainProgressTime = gameTime();
            brainLastResources = resources();
            brainLastArmy = mainArmyCount();
            brainLastBuildings = buildingsConstructed();
            brainLastKnownNodes = brainKnownUsefulNodes;
            brainStalled = false;
        } else {
            brainStalled =
                gameTime() - brainProgressTime > 14;
        }
    }

    function brainScoreGoals() {
        scoreDefend = brainThreatLevel * 85;

        // A single worker (or a remote outpost) being harassed already gets its
        // own response every tick no matter what brainGoal is active — see
        // evacuateShapers()/builderCargoSafety()/positionEconomyGuards() in
        // brainExecuteGoal(). It doesn't also need the whole army recalled off
        // an attack that's already under way. Only a threat actually at the
        // main base (or the base itself taking hits) is serious enough to
        // override everything else — that's the only case that earns the full
        // emergency bonus. Anything smaller still nudges Defend up a little via
        // brainThreatLevel and this reduced bonus, without being able to
        // single-handedly outscore and abort an in-progress attack.
        if (commandCenterUnderAttack() || baseThreat > 0) {
            scoreDefend = scoreDefend + 900;
        } else if (emergencyActive) {
            scoreDefend = scoreDefend + 200;
        }

        if (commandCenterUnderAttack()) {
            scoreDefend = scoreDefend + 700;
        }

        scoreEconomy = 180 - brainEconomyHealth;

        if (economicWarning) {
            scoreEconomy = scoreEconomy + 260;
        }

        if (economicCrisis) {
            scoreEconomy = scoreEconomy + 650;
        }

        scoreScout = 100 + brainKnowledgeNeed;

        if (unitCount("Wisp") <= 0) {
            scoreScout = scoreScout + 180;
        }

        scoreExpand = -9999;

        if (brainCandidateNode != "") {
            scoreExpand =
                170 +
                expansionUtility(brainCandidateNode) -
                brainThreatLevel * 35;

            if (economicWarning) {
                scoreExpand = scoreExpand + 180;
            }
        }

        scoreRecover =
            recentLosses * 45 +
            damagedArmyCount * 12;

        if (armyHealthScore < 0.55) {
            scoreRecover = scoreRecover + 350;
        }

        if (phase == 3) {
            scoreRecover = scoreRecover + 120;
        }

        scoreAttack =
            120 +
            brainArmyReadiness * 3 +
            max(0, brainEconomyHealth);

        if (
            ((phase != 2 && phase != 4) && !readyToAttack()) ||
            currentPower <= 0 ||
            (phase!=2 && phase!=4 && attackMinimum()>=requiredArmy && frontlineCount()<2)
        ) {
            scoreAttack = -9999;
        }

        if (
            pressureObjective == "" && !enemyBaseFound() && enemyBaseDistance < 120 &&
            brainVisibleBuildings <= 0 &&
            count(visibleEnemyUnits()) <= 0
        ) {
            scoreAttack = -9999;
            scoreScout = scoreScout + 420;
        }

        // Same distinction as scoreDefend above: a threat at the base is worth
        // seriously discouraging an attack over; a harassed worker or remote
        // asset elsewhere on the map (which already gets its own response) isn't.
        if (commandCenterUnderAttack() || baseThreat > 0) {
            scoreAttack = scoreAttack - 300;
        } else if (brainThreatLevel > 0) {
            scoreAttack = scoreAttack - 80;
        }

        if (economicCrisis) {
            scoreAttack = scoreAttack - 500;
        }

        // Keep the offensive goal while the deployed force is healthy.
        // requiredArmy rises after launch; that must not cancel this wave.
        if ((phase == 2 || phase == 4) && !economicCrisis &&
            baseThreat <= 0 && !commandCenterUnderAttack() && !armyFightRetreat) {
            scoreAttack = scoreAttack + 450;
        }
        if (phase == 2 && armyHealthScore >= 0.55 &&
            recentLosses < 4 && !armyFightRetreat) {
            scoreRecover = -9999;
        }

        scoreCleanup = -9999;

        if (brainVisibleBuildings > 0) {
            scoreCleanup =
                240 +
                brainArmyReadiness * 3;
        }

        if (cleanupMode) {
            scoreCleanup = scoreCleanup + 220;
        }

        scoreTech = 160;

        if (hasBuilding("Memory Bloom")) {
            scoreTech = scoreTech + 90;
        }

        if (safeToResearch) {
            scoreTech = scoreTech + 100;
        }

        if (economicWarning) {
            scoreTech = scoreTech - 220;
        }

        scoreBuild = 210 - brainArmyReadiness;

        if (!hasBuilding("Spawning Vault")) {
            scoreBuild = scoreBuild + 700;
        }

        if (!hasBuilding("Titan Crucible")) {
            scoreBuild = scoreBuild + 180;
        }

        if (mainArmyCount() < 8) {
            scoreBuild = scoreBuild + 260;
        }

        scoreNaval = -9999;

        if (waterAvailable) {
            scoreNaval = 120 + navyCount() * 8;

            if (!hasBuilding("Tidewomb")) {
                scoreNaval = scoreNaval + 110;
            }

            // No land path to anyone at all (see isIsolatedByWater()) — nothing
            // else matters until there's a way off this island, so this goal
            // needs to actually win outright, not just compete — including
            // against a full economicCrisis + economicWarning stack on
            // scoreEconomy (up to ~910 plus stickiness), since a resource
            // pinch is still secondary to being unable to reach anyone at all.
            if (
                isIsolated &&
                !hasBuilding("Tidewomb")
            ) {
                scoreNaval = scoreNaval + 1300;
            }
        }
    }

    function brainChooseGoal() {
        brainBestGoal = 0;
        brainBestScore = -99999;

        brainConsider(1, scoreDefend);
        brainConsider(2, scoreEconomy);
        brainConsider(3, scoreExpand);
        brainConsider(4, scoreScout);
        brainConsider(5, scoreBuild);
        brainConsider(6, scoreTech);
        brainConsider(7, scoreAttack);
        brainConsider(8, scoreRecover);
        brainConsider(9, scoreCleanup);
        brainConsider(10, scoreNaval);

        if (
            brainGoal == 0 ||
            brainBestGoal != brainGoal
        ) {
            brainPreviousGoal = brainGoal;
            brainGoal = brainBestGoal;
            brainGoalStarted = gameTime();
            brainProgressTime = gameTime();
        }

        brainGoalScore = brainBestScore;
    }

        if (!hasBuilding("Spawning Vault")) {
            if (unitCount("Shaper") < 3) {
                produceShaper();
            } else {
                safeConstruct("Spawning Vault");
            }

            return;
        }

        manageConstruction();
    }

    function brainExecuteDefend() {
        if (!fullHomeDefense) {
            // A local interception owns its squad, not the whole attacking force.
            produceEmergencyDefenders();
            if (interceptActive || homeDanger) { commandHomeDefense(); }
            positionForwardDefense();
            return;
        }
        phase = 3;
        // Fighting owns the army until the threat is gone. Force the subsequent
        // recovery pass to choose a fresh shared rally instead of reusing one
        // from an older battle.
        regroupActive = false;
        cleanupMode = false;
        finishing = false;
        groupingReadySince = -1;
        defendMainArmy(defenseX, defenseY);

        if (commandCenterUnderAttack()) {
            defendMainArmy(defenseX,defenseY);
        }

        produceBarracksUnit();
        produceFactoryUnit();
    }

    }

    function brainExecuteTech() {
        brainEnsureInfrastructure();
        manageResearch();

        // Was gated at a fixed "< 10" instead of the real army target — once the
        // army reached 10 (well under both desiredArmyCap, up to 64, and
        // requiredArmy/attackMinimum(), 14+), Tech stopped training entirely.
        // Since Tech routinely outscores Attack/Build/Economy once buildings and
        // research needs stabilize, the AI got stuck producing nothing forever
        // with an army permanently below the threshold needed to ever attack —
        // resources piling up unspent while nothing happens. produceBarracksUnit()/
        // produceFactoryUnit() already self-gate via armyProductionCapped()
        // (desiredArmyCap) and shouldSaveForExpansion()/shouldSaveForResearch(),
        // so calling them here unconditionally doesn't overspend — it just lets
        // Tech keep building the army toward its actual target like every other
        // goal does.
        produceBarracksUnit();
        produceFactoryUnit();
    }

    function brainExecuteAttack() {
        produceBarracksUnit();
        produceFactoryUnit();

        if (
            currentPower <= 0 ||
            mainArmyCount() < 3
        ) {
            beginRecovery();
            return;
        }

        if (
            pressureObjective == "" && !enemyBaseFound() && enemyBaseDistance < 120 &&
            count(visibleEnemyBuildings()) <= 0 &&
            count(visibleEnemyUnits()) <= 0
        ) {
            phase = 1;
            regroupActive = false;
            safeScoutMap();
            return;
        }

        if (phase != 2 && phase != 4) {
            if (readyToAttack()) {
                beginGrouping();
            } else {
                brainExecuteBuild();
                return;
            }
        }

        if (phase == 4) {
            if (attackMinimum()>groupingMinimum && groupingMinimum<requiredArmy) {
                phase=1; groupingReadySince=-1; brainExecuteBuild(); return;
            }
            commandGrouping();

            if (groupingReady()) {
                launchAttack();
            }
        }

        if (phase == 2) {
            if (
                attackCollapsed() ||
                (objectiveStalled && count(marchingHeld) <= 0 &&
                    gameTime() - objectiveProgressTime > 14)
            ) {
                if (!battleOutcomeRecorded) {
                    recordBattleOutcome(false);
                    battleOutcomeRecorded = true;
                }
                beginRecovery();
            } else {
                setPressureDestination();
                executeObjectiveDecision();
            }
        }
    }

    function brainExecuteRecover() {
        phase = 3;
        cleanupMode = false;
        finishing = false;
        groupingReadySince = -1;
        regroupMainArmy();
        produceBarracksUnit();
        produceFactoryUnit();

        // Was 0.72/<3: waiting for near-full health before attacking again gives
        // the enemy the same recovery time back every single cycle. Re-engaging
        // once the army is merely "healthy enough" (not fully healed) keeps
        // pressure on instead of both sides resetting to full strength forever.
        if (
            gameTime() >= regroupUntil &&
            armyHealthScore >= 0.58 &&
            recentLosses < 5
        ) {
            phase = 1;
            regroupActive = false;
        }
    }

    function brainExecuteCleanup() {
        if (phase != 2) {
            brainExecuteAttack();
            return;
        }
        cleanupMode = true;
        runBuildingCleanup();
    }

    function brainExecuteNaval() {
        brainEnsureInfrastructure();
        manageResearch();
        produceNavy();
        commandNavy();
    }

    // Runs LAST, after attack/cleanup/micro orders, so fast units cannot
    // immediately overwrite their wait order and race away again.
    function marchingContains(list, wanted) {
        for (member in list) {
            if (member == wanted) { return true; }
        }
        return false;
    }

    function maintainMarchCohesion() {
        let slowestSpeed = 999999;
        let nextHeld = [];
        let goalX = 0;
        let goalY = 0;
        let anchorDistance = 0;
        let separation = 0;
        let ahead = false;
        let danger = "";
        let waiting = false;
        let joined = [];
        let maxSeparation = 0;

        // Explicit assignments are intentional: this language persists lets.
        slowestSpeed = 999999;
        nextHeld = [];
        joined = [];
        maxSeparation = 0;
        marchingAnchor = "";
        if (phase != 2 || emergencyActive || decisionMode == -1) {
            marchingHeld = [];
            marchingRegrouping = false;
            return false;
        }
        goalX = objectiveX;
        goalY = objectiveY;
        if (cleanupMode && cleanupTarget != "") {
            goalX = cleanupTargetX;
            goalY = cleanupTargetY;
        }

        // Never use fresh units still at home as the pace setter.
        for (id in marchingWave) {
            if (unitExists(id) && !isDefenseReserved(id) && !localEscapeOwned(id) && !marchingContains(raidingUnits,id) && !(homeDanger && marchingContains(homeDefenseUnits, id)) &&
                unitHealthRatio(id) >= 0.46 && unitOrder(id) != "retreat" && unitSpeed(id) > 0) {
                joined[count(joined)] = id;
                if (unitSpeed(id) < slowestSpeed) {
                    slowestSpeed = unitSpeed(id);
                    marchingAnchor = id;
                }
            }
        }
        marchingWave = joined;
        if (marchingAnchor == "") { marchingRegrouping = false; return false; }
        anchorDistance = distanceBetween(unitX(marchingAnchor), unitY(marchingAnchor), goalX, goalY);

        // Whole-column check, not just neighbor-to-neighbor: the per-unit
        // ahead/behind logic below only ever compares each member against the
        // anchor by a small per-tick threshold, so a long march at uneven
        // speeds can still drift into a column stretched the length of the map
        // -- every individual check quietly satisfied at every instant, yet the
        // head and tail end up nowhere near each other. Measuring the actual
        // furthest straggler's distance from the anchor catches that and forces
        // the whole wave to halt and bunch back up before continuing, instead
        // of only ever nudging one laggard at a time.
        for (id in marchingWave) {
            if (id != marchingAnchor && !localEscapeOwned(id) && unitOrder(id) != "retreat") {
                danger = nearestEnemyUnitTo(unitX(id), unitY(id), unitSight(id) + 35);
                if (danger == "" || !inRange(unitX(id), unitY(id), unitX(danger), unitY(danger),
                    max(unitAttackRange(id), unitAttackRange(danger)) + 20)) {
                    separation = distanceBetween(unitX(id), unitY(id),
                        unitX(marchingAnchor), unitY(marchingAnchor));
                    if (separation > maxSeparation) { maxSeparation = separation; }
                }
            }
        }
        // Hysteresis: a wide gap forces a full halt; it only lifts once the
        // column has actually tightened back up, not the instant the gap dips
        // under the same number it tripped at (which would just flap on and
        // off every tick right at the boundary).
        if (marchingRegrouping) {
            if (maxSeparation <= 90) { marchingRegrouping = false; }
        } else {
            if (maxSeparation > 170) { marchingRegrouping = true; }
        }

        for (id in marchingWave) {
            separation = distanceBetween(unitX(id), unitY(id),
                unitX(marchingAnchor), unitY(marchingAnchor));
            ahead = distanceBetween(unitX(id), unitY(id), goalX, goalY) < anchorDistance;
            waiting = marchingContains(marchingHeld, id);
            danger = nearestEnemyUnitTo(unitX(id), unitY(id), unitSight(id) + 35);

            // Preserve fighting and escape orders when already in combat.
            if (danger == "" || !inRange(unitX(id), unitY(id), unitX(danger), unitY(danger),
                max(unitAttackRange(id), unitAttackRange(danger)) + 20)) {
                if (marchingRegrouping && id != marchingAnchor) {
                    // Whole column bunching up: anyone still far closes the gap,
                    // anyone already close holds here instead of trickling
                    // further ahead while the rest catch up.
                    if (separation > 70) {
                        guardedMoveUnit(id, unitX(marchingAnchor), unitY(marchingAnchor));
                    } else {
                        guardedDefendUnit(id, unitX(marchingAnchor), unitY(marchingAnchor));
                    }
                    nextHeld[count(nextHeld)] = id;
                } else {
                    if (id != marchingAnchor && ahead &&
                        (separation > 75 || (waiting && separation > 45))) {
                        // Stop the faster member until the slow member catches up.
                        guardedStopUnit(id);
                        nextHeld[count(nextHeld)] = id;
                    } else {
                        if (id != marchingAnchor && !ahead && separation > 100) {
                            guardedMoveUnit(id, unitX(marchingAnchor), unitY(marchingAnchor));
                        } else {
                            if (waiting && (decisionMode == 1 || cleanupMode)) { guardedAttackMove(id, goalX, goalY); }
                        }
                    }
                }
            }
        }
        marchingHeld = nextHeld;
        if (count(marchingHeld) > 0) {
            // Deliberate waiting is not a failed path / stalled attack.
            objectiveProgressTime = gameTime();
            objectiveStalled = false;
            decisionUntil = max(decisionUntil, gameTime() + 5);
        }

        // Reinforcements catch the travelling group before joining its roster.
        for (id in mainFighters()) {
            if (unitExists(id) && !isDefenseReserved(id) && !localEscapeOwned(id) && !marchingContains(raidingUnits,id) && !(homeDanger && marchingContains(homeDefenseUnits, id)) &&
                unitHealthRatio(id) >= 0.46 && unitOrder(id) != "retreat" && !marchingContains(marchingWave, id)) {
                if (inRange(unitX(id), unitY(id), unitX(marchingAnchor), unitY(marchingAnchor), 110)) {
                    marchingWave[count(marchingWave)] = id;
                } else {
                    danger = nearestEnemyUnitTo(unitX(id), unitY(id), unitSight(id) + 35);
                    if (danger == "") {
                        guardedMoveUnit(id, unitX(marchingAnchor), unitY(marchingAnchor));
                    }
                }
            }
        }
        return true;
    }

    // Requires the updated engine API: resumeConstruction(buildingId),
    // buildingHasBuilder(buildingId). Assignment is requested by this AI.
    function rescueAbandonedConstruction() {
        let rescued = 0;
        let needsBuilder = false;
        rescued = 0;
        needsBuilder = false;
        if (gameTime() < nextConstructionRescue) { return 0; }
        nextConstructionRescue = gameTime() + 2;

        for (building in myBuildings()) {
            if (buildingExists(building) && buildingUnderConstruction(building) &&
                !buildingHasBuilder(building) &&
                safeConstructionSite(buildingX(building), buildingY(building))) {
                if (resumeConstruction(building)) {
                    rescued = rescued + 1;
                    // Give a replacement travelling to the expansion time to arrive.
                    if (pendingExpansionNode != "" && inRange(
                        buildingX(building), buildingY(building),
                        pendingExpansionX, pendingExpansionY, expansionCoverageRadius)) {
                        pendingExpansionProgressTime = gameTime();
                        pendingExpansionUntil = gameTime() + 35;
                    }
                } else {
                    needsBuilder = true;
                }
            }
        }
        if (needsBuilder) {
            // Existing production limits/queue checks prevent builder spam.
            produceShaper();
        }
        return rescued;
    }

    // Safety and defense take precedence over the economic goal scores.
    function siteThreatPower(x, y, buffer) {
        let risk = 0;
        risk = 0;
        for (enemy in visibleEnemyUnits()) {
            if (unitExists(enemy) && unitAttack(enemy) > 0 && inRange(
                x, y, unitX(enemy), unitY(enemy), unitAttackRange(enemy) + buffer)) {
                risk = risk + unitAttack(enemy) + 3;
            }
        }
        for (enemy in visibleEnemyBuildings()) {
            if (buildingExists(enemy) && buildingAttack(enemy) > 0 && inRange(
                x, y, buildingX(enemy), buildingY(enemy), buildingAttackRange(enemy) + buffer)) {
                risk = risk + buildingAttack(enemy) + 8;
            }
        }
        return risk;
    }

    function safeConstructionSite(x, y) {
        if (homeDanger || siteThreatPower(x, y, 85) > 0) { return false; }
        for (contact in recentEnemyContacts) {
            if (gameTime() - contact[4] < 8 && inRange(x, y, contact[1], contact[2], 190)) {
                return false;
            }
        }
        return true;
    }

    function safeConstruct(type) {
        // Auto-placement is near home; require a clear base perimeter.
        if (!safeConstructionSite(baseX(), baseY()) ||
            siteThreatPower(baseX(), baseY(), 300) > 0) { return false; }
        return construct(type);
    }
    function safeConstructAt(type, x, y) {
        if (!safeConstructionSite(x, y)) { return false; }
        return constructAt(type, x, y);
    }
    function safeExpandToNode(node) {
        if (!nodeExists(node) || !safeConstructionSite(nodeX(node), nodeY(node))) { return false; }
        return expandToNode(node);
    }
    function safeResearch(tech) {
        if (homeDanger) { return false; }
        return research(tech);
    }

    function assessHomeDefense() {
        let bestScore = -1;
        let score = 0;
        let nearestAsset = "";
        let nearestDistance = 0;
        let distance = 0;
        let immediate = false;
        let candidates = [];
        let best = "";
        let bestDistance = 0;
        let selecting = true;
        homeThreatCount=0; homeThreatPower=0; homeThreatTarget=""; homeThreatBuilding="";
        homeDefenseUnits=[]; homeDefensePower=0; interceptActive=false;
        bestScore=-1; immediate=false;
        for (enemy in visibleEnemyUnits()) {
            if (unitExists(enemy) && unitAttack(enemy)>0) {
                nearestAsset=""; nearestDistance=999999;
                for (building in myBuildings()) {
                    if (buildingExists(building)) {
                        distance=distanceBetween(unitX(enemy),unitY(enemy),buildingX(building),buildingY(building));
                        if (distance<nearestDistance) { nearestDistance=distance; nearestAsset=building; }
                    }
                }
                if (nearestAsset!="" && nearestDistance<=unitAttackRange(enemy)+max(220,min(320,unitSpeed(enemy)*4))) {
                    interceptActive=true;
                    homeThreatCount=homeThreatCount+1;
                    homeThreatPower=homeThreatPower+unitAttack(enemy)+3;
                    if (nearestDistance<=unitAttackRange(enemy)+65) { immediate=true; }
                    score=unitAttack(enemy)*2+unitAttackRange(enemy)/5-nearestDistance*0.1;
                    if (nearestDistance<=unitAttackRange(enemy)+30) { score=score+100; }
                    if (score>bestScore) {
                        bestScore=score; homeThreatTarget=enemy;
                        homeThreatX=unitX(enemy); homeThreatY=unitY(enemy);
                        defenseAssetX=buildingX(nearestAsset); defenseAssetY=buildingY(nearestAsset);
                    }
                }
            }
        }
        for (enemy in visibleEnemyBuildings()) {
            if (buildingExists(enemy) && buildingAttack(enemy)>0) {
                for (building in myBuildings()) {
                    if (buildingExists(building) && inRange(buildingX(building),buildingY(building),
                        buildingX(enemy),buildingY(enemy),buildingAttackRange(enemy)+80)) {
                        immediate=true; interceptActive=true; homeThreatBuilding=enemy;
                        homeThreatPower=homeThreatPower+buildingAttack(enemy)+8;
                        if (homeThreatTarget=="") {
                            homeThreatX=buildingX(enemy); homeThreatY=buildingY(enemy);
                            defenseAssetX=buildingX(building); defenseAssetY=buildingY(building);
                        }
                    }
                }
            }
        }
        if (immediate || commandCenterUnderAttack()) { homeDefenseUntil=gameTime()+8; }
        homeDanger=gameTime()<homeDefenseUntil;
        fullHomeDefense=false;
        if (!interceptActive && !homeDanger) { return; }
        if (!interceptActive) {
            defenseAssetX=fieldRallyX; defenseAssetY=fieldRallyY;
            homeThreatX=fieldRallyX; homeThreatY=fieldRallyY;
        }
        distance=max(1,distanceBetween(homeThreatX,homeThreatY,defenseAssetX,defenseAssetY));
        interceptX=clamp(homeThreatX+(defenseAssetX-homeThreatX)*min(60,distance*0.2)/distance,28,mapWidth()-28);
        interceptY=clamp(homeThreatY+(defenseAssetY-homeThreatY)*min(60,distance*0.2)/distance,28,mapHeight()-28);
        candidates=mainFighters(); selecting=true;
        while (selecting && count(homeDefenseUnits)<6 &&
            (homeDefensePower<homeThreatPower*1.6 || count(homeDefenseUnits)<2)) {
            best=""; bestDistance=999999;
            for (id in candidates) {
                if (unitExists(id) && !localEscapeOwned(id) && unitHealthRatio(id)>=0.4 &&
                    !marchingContains(homeDefenseUnits,id)) {
                    distance=distanceBetween(unitX(id),unitY(id),interceptX,interceptY);
                    if (distance<bestDistance) { bestDistance=distance; best=id; }
                }
            }
            if (best=="") { selecting=false; }
            else { homeDefenseUnits[count(homeDefenseUnits)]=best; homeDefensePower=homeDefensePower+unitAttack(best)+3; }
        }
        fullHomeDefense=homeDanger && (homeThreatCount>=3 || homeThreatBuilding!="" ||
            homeDefensePower<homeThreatPower*1.3 || count(homeDefenseUnits)>=max(2,mainArmyCount()-3));
        for (hq in myBuildingsOfType("Heartspire")) {
            if (buildingExists(hq) && buildingHealthRatio(hq)<0.6 && commandCenterUnderAttack()) { fullHomeDefense=true; }
        }
    }

    function produceEmergencyDefenders() {
        if (gameTime() < nextEmergencyProduction) { return; }
        nextEmergencyProduction = gameTime() + 2;
        for (producer in myBuildingsOfType("Spawning Vault")) {
            if (buildingExists(producer) && !buildingUnderConstruction(producer) &&
                buildingQueueLength(producer) < 1) {
                if (frontlineCount() < 4) { train("soldier", producer); }
                else {
                    if (!train("marksman", producer)) { train("soldier", producer); }
                }
            }
        }
        for (producer in myBuildingsOfType("Titan Crucible")) {
            if (buildingExists(producer) && !buildingUnderConstruction(producer) &&
                buildingQueueLength(producer) < 1 && resources() >= unitCost("tank")) {
                train("tank", producer);
            }
        }
    }

    function commandHomeDefense() {
        issuingDefenseOrders=true;
        for (id in homeDefenseUnits) {
            if (unitExists(id)) {
                if (unitHealthRatio(id)<0.25) { localRetreatUnit(id); }
                else {
                    if (interceptActive) { guardedAttackMove(id,interceptX,interceptY); }
                    else { guardedDefendUnit(id,fieldRallyX,fieldRallyY); }
                }
            }
        }
        issuingDefenseOrders=false;
    }

    function workerEscaping(id) {
        for (worker in myUnitsOfType("Shaper")) {
            if (worker == id && unitExists(id)) {
                return localEscapeOwned(id) || siteThreatPower(unitX(id),unitY(id),65) > 0;
            }
        }
        return false;
    }

    function workerEscapeRisk(x,y) { return localThreatRisk(x,y); }

    function protectWorkersFromDanger() {
        let observations = [];
        let lastHp = 0;
        observations = [];
        for (worker in myUnitsOfType("Shaper")) {
            if (unitExists(worker)) {
                lastHp = unitHp(worker);
                for (old in workerEscapeMemory) {
                    if (old[0] == worker) { lastHp = old[1]; }
                }
                if (unitHp(worker) < lastHp) {
                    // Damage is evidence of danger even when the shooter is outside vision.
                    requestLocalEscape(worker,4);
                } else {
                    if (siteThreatPower(unitX(worker),unitY(worker),65) > 0) {
                    requestLocalEscape(worker,3);
                    }
                }
                observations[count(observations)] = [worker,unitHp(worker)];
            }
        }
        workerEscapeMemory = observations;
    }

    function executeHomeEmergency() {
        if (economyRaidTarget != "") { endEconomyRaid(); }
        cleanupMode = false;
        finishing = false;
        groupingReadySince = -1;
        phase = 3;
        regroupActive = false;
        regroupUntil = max(regroupUntil, gameTime() + 8);
        if (fullHomeDefense) { homeDefenseUnits = mainFighters(); }
        produceEmergencyDefenders();
        commandHomeDefense();
        manageCombatSkills();
        protectWisps();
    }

    function brainExecuteGoal() {
        if (homeDanger && (fullHomeDefense || phase != 2)) {
            decisionReason = "Defend home: protect workers and produce fighters";
            executeHomeEmergency();
            return;
        }
        rescueAbandonedConstruction();
        // Research is a background production lane. A long attack/economy goal
        // must not prevent the lab from starting its next valid technology.
        manageResearch();
        assignMineNetwork();
        builderCargoSafety();
        protectWisps();

        if (brainGoal == 1) {
            brainExecuteDefend();
        }

        if (brainGoal == 2) {
            brainExecuteEconomy();
        }

        if (brainGoal == 3) {
            brainExecuteExpand();
        }

        if (brainGoal == 4) {
            brainExecuteScout();
        }

        if (brainGoal == 5) {
            brainExecuteBuild();
        }

        if (brainGoal == 6) {
            brainExecuteTech();
        }

        if (brainGoal == 7) {
            brainExecuteAttack();
        }

        if (brainGoal == 8) {
            brainExecuteRecover();
        }

        if (brainGoal == 9) {
            brainExecuteCleanup();
        }

        if (brainGoal == 10) {
            brainExecuteNaval();
        }

        runCombatMicro();
        manageCombatSkills();
        rotateDamagedUnits();
        maintainMarchCohesion();
        rejoinIdleFighters();
        pressureExposedWorkers();
        // Phase 3 used to make every survivor choose a separate local escape on
        // every tick. Keep the common regroup order alive even if the strategic
        // director temporarily chooses Economy/Build/Tech after the battle.
        if (phase == 3 && !homeDanger && brainGoal != 1) {
            regroupMainArmy();
        }
        if (phase == 4 && !emergencyActive) {
            commandGrouping();
            if (groupingReady()) { launchAttack(); }
        }
    }

    function brainThink() {
        brainObserve();
        updateForwardRally();
        updateStandingDefense();
        assessHomeDefense();
        updatePressureObjective();
        brainScoreGoals();
        brainChooseGoal();
        if (!fullHomeDefense && !armyFightRetreat && armyHealthScore>=0.6 &&
            (phase==2 || phase==4 || (pressureObjective!="" && readyToAttack())) && !economicCrisis) {
            brainGoal=7;
        }
        if (homeDanger && !fullHomeDefense && phase == 2) {
            brainGoal = 7;
            decisionReason = "Small raid: send nearby defenders; keep main attack";
        } else {
            decisionReason = "Normal strategy: economy, formation and attack readiness";
        }
        let savedEmergency = false;
        savedEmergency = emergencyActive;
        if ((homeDanger || interceptActive) && !fullHomeDefense && phase == 2) { emergencyActive = false; }
        if (interceptActive && !homeDanger) { decisionReason="Intercept approaching enemies before they reach buildings"; }
        if (brainGoal==7 && pressureObjective!="" && !homeDanger) { decisionReason="Advance on the scouted enemy economy"; }
        brainExecuteGoal();
        emergencyActive = savedEmergency;
        // The defense squad and worker escape have final command priority.
        positionForwardDefense();
        if (interceptActive || homeDanger) { commandHomeDefense(); }
        if (decisionReason != lastDecisionReason && gameTime() >= nextDecisionLog) {
            log(decisionReason);
            lastDecisionReason = decisionReason;
            nextDecisionLog = gameTime() + 5;
        }
    }

    brainThink();

    // Top-level recovery remains effective when a running match has retained
    // older function definitions. It prevents a depleted economy from freezing
    // forever after every previously covered crystal node disappears.
    if (
        economicCrisis &&
        productiveCoveredNodes <= 0 &&
        count(knownResourceNodes()) > 0 &&
        gameTime() >= hotRecoveryExpansionTick &&
        pendingExpansionNode == "" &&
        resources() >= 320 &&
        !emergencyActive && !homeDanger
    ) {
        hotRecoveryNode = "";
        hotRecoveryScore = 999999;

        for (node in knownResourceNodes()) {
            if (
                nodeExists(node) &&
                !nodeCovered(node) &&
                nodeRemaining(node) > 250 &&
                count(enemyUnitsNear(nodeX(node), nodeY(node), 220)) <= 0 &&
                count(enemyBuildingsNear(nodeX(node), nodeY(node), 270)) <= 0 &&
                !buildingNear("Whisper Node", nodeX(node), nodeY(node), 320)
            ) {
                hotRecoveryCandidateScore =
                    distanceBetween(baseX(), baseY(), nodeX(node), nodeY(node)) -
                    min(1800, nodeRemaining(node)) / 3;

                if (hotRecoveryCandidateScore < hotRecoveryScore) {
                    hotRecoveryScore = hotRecoveryCandidateScore;
                    hotRecoveryNode = node;
                }
            }
        }

        if (hotRecoveryNode != "") {
            phase = 1;
            decisionMode = 0;
            objectiveStalled = false;
            brainCandidateNode = hotRecoveryNode;
            expansionCandidate = hotRecoveryNode;
            expansionCandidateX = nodeX(hotRecoveryNode);
            expansionCandidateY = nodeY(hotRecoveryNode);
            expansionDanger = 0;
            guardedDefendUnits(mainFighters(), expansionCandidateX, expansionCandidateY);

            if (safeExpandToNode(hotRecoveryNode)) {
                registerPendingExpansion(hotRecoveryNode);
                expansionSavingsActive = false;
                expansionSavingsNode = "";
                hotRecoveryExpansionTick = gameTime() + 25;
            } else {
                hotRecoveryExpansionTick = gameTime() + 3;
            }
        } else {
            safeScoutMap();
            hotRecoveryExpansionTick = gameTime() + 6;
        }
    }


    // Final order priority: scouting/economic fallbacks cannot undo safety.
    protectWisps();

    // Keep final economic/scouting fallbacks from cancelling worker escape.

    // Single final worker safety pass: preserves paths and wins over economic orders.
    protectWorkersFromDanger();

    // Only this final pass issues retreat/kiting moves; later orders cannot undo it.
    executeLocalEscapes();
